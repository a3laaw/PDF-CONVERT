   111→async function renderPageDataUrl(file: File, pageNo: number, maxW = 820): Promise<string> {
   112→  const pdfjs = await import("pdfjs-dist");
   113→  pdfjs.GlobalWorkerOptions.workerSrc = "/pdf.worker.min.mjs";
   114→  const bytes = new Uint8Array(await file.arrayBuffer());
   115→  const doc = await pdfjs.getDocument({ data: bytes }).promise;
   116→  const page = await doc.getPage(Math.max(1, pageNo));
   117→  const v1 = page.getViewport({ scale: 1 });
   118→  const viewport = page.getViewport({ scale: Math.min(1.8, maxW / v1.width) });
   119→  const canvas = document.createElement("canvas");
   120→  canvas.width = Math.floor(viewport.width);
   121→  canvas.height = Math.floor(viewport.height);
   122→  const c2 = canvas.getContext("2d")!;
   123→  c2.fillStyle = "#ffffff";
   124→  c2.fillRect(0, 0, canvas.width, canvas.height);
   125→  await page.render({ canvas, canvasContext: c2, viewport } as never).promise;
   126→  return canvas.toDataURL("image/jpeg", 0.85);
   127→}

   129→function fmtSize(b: number): string {
   130→  if (b >= 1048576) return `${toHindiDigits((b / 1048576).toFixed(1).replace(".", "٫"))} م.ب`;
   131→  return `${toHindiDigits(Math.max(1, Math.round(b / 1024)))} ك.ب`;
   132→}
   133→
   134→function extLabel(name: string): string {
   135→  const e = name.match(/\.([^.]+)$/)?.[1]?.toUpperCase() ?? "ملف";
   136→  return e === "JPG" ? "JPG" : e;
   137→}

   141→export function Workbench({ onOpenConvert }: { onOpenConvert: () => void }) {
   142→  const files = useWorkspace((s) => s.files);
   143→  const results = useWorkspace((s) => s.results);
   144→  const runState = useWorkspace((s) => s.runState);
   145→  const runMessage = useWorkspace((s) => s.runMessage);
   146→  const runProgress = useWorkspace((s) => s.runProgress);
   147→  const addFiles = useWorkspace((s) => s.addFiles);
   148→  const removeFile = useWorkspace((s) => s.removeFile);
   149→  const runOnFiles = useWorkspace((s) => s.runOnFiles);
   150→  const renameResult = useWorkspace((s) => s.renameResult);
   151→  const clearResults = useWorkspace((s) => s.clearResults);
   152→  const downloadAll = useWorkspace((s) => s.downloadAll);
   153→
   154→  const [activeId, setActiveId] = useState<string | null>(null);
   155→  const [selFiles, setSelFiles] = useState<Set<string> | null>(null);
   156→  const [tool, setTool] = useState<OpId | null>(null);
   157→  const [params, setParams] = useState<Record<string, string>>({});
   158→  const [selPages, setSelPages] = useState<number[]>([]);
   159→  const [thumbs, setThumbs] = useState<Record<string, string[]>>({});
   160→  const [page, setPage] = useState(1);
   161→
   162→  const [renaming, setRenaming] = useState<{ id: string; value: string } | null>(null);
   163→  const [dragOver, setDragOver] = useState(false);
   164→  const inputRef = useRef<HTMLInputElement>(null);
   165→  const thumbsReq = useRef<Set<string>>(new Set());
   166→
   167→  /* الملف النشط: المعيّن بالنقر أو آخر ملف أُضيف — اشتقاق بلا effects */
   168→  const activeFile = files.find((f) => f.id === activeId) ?? files[files.length - 1] ?? null;
   169→  const effectiveSel = selFiles ?? (activeFile ? new Set([activeFile.id]) : new Set<string>());
   170→  const running = runState === "running";
   171→
   172→  /* مصغرات صفحات الـ PDF النشط */
   173→  useEffect(() => {
   174→    if (!activeFile || activeFile.kind !== "pdf") return;
   175→    if (thumbs[activeFile.id] || thumbsReq.current.has(activeFile.id)) return;
   176→    thumbsReq.current.add(activeFile.id);
   177→    pdfPageThumbs(activeFile.file, 60)
   178→      .then((arr) => setThumbs((t) => ({ ...t, [activeFile.id]: arr })))
   179→      .catch(() => thumbsReq.current.delete(activeFile.id));
   180→  }, [activeFile, thumbs]);
   181→
   182→  /* المعاينة الكبيرة للصفحة النشطة — مفتوحة بالمفتاح لتفادي setState المتزامن */
   183→  const previewKey = activeFile ? `${activeFile.id}:${activeFile.kind === "image" ? "img" : page}` : "";
   184→  const [previewState, setPreviewState] = useState<{ key: string; url: string } | null>(null);
   185→  useEffect(() => {
   186→    if (!activeFile || !previewKey) return;
   187→    let alive = true;
   188→    let objUrl: string | null = null;
   189→    const done = (url: string) => {
   190→      if (alive) setPreviewState({ key: previewKey, url });
   191→    };
   192→    if (activeFile.kind === "image") {
   193→      Promise.resolve().then(() => {
   194→        objUrl = URL.createObjectURL(activeFile.file);
   195→        done(objUrl);
   196→      });
   197→    } else {
   198→      renderPageDataUrl(activeFile.file, page)
   199→        .then(done)
   200→        .catch(() => undefined);
   201→    }
   202→    return () => {
   203→      alive = false;
   204→      if (objUrl) URL.revokeObjectURL(objUrl);
   205→    };
   206→  }, [activeFile, page, previewKey]);
   207→  const preview = previewState && previewState.key === previewKey ? previewState.url : "";
   208→
   209→  const pickTool = (id: OpId | "convert") => {
   210→    if (id === "convert") {
   211→      onOpenConvert();
   212→      return;
   213→    }
   214→    setTool(id);
   215→    const def: Record<string, string> = {};
   216→    for (const f of TOOLS_BY_ID[id].fields) if (f.default !== undefined) def[f.key] = f.default;
   217→    const pf = TOOLS_BY_ID[id].fields.find((f) => f.pagesPicker);
   218→    if (pf) def[pf.key] = selPages.join(",");
   219→    setParams(def);
   220→  };
   221→
   222→  const togglePage = (n: number) => {
   223→    const next = selPages.includes(n)
   224→      ? selPages.filter((x) => x !== n)
   225→      : [...selPages, n].sort((a, b) => a - b);
   226→    setSelPages(next);
   227→    if (tool) {
   228→      const pf = TOOLS_BY_ID[tool].fields.find((f) => f.pagesPicker);
   229→      if (pf) setParams((p) => ({ ...p, [pf.key]: next.join(",") }));
   230→    }
   231→  };
   232→
   233→  const toggleSel = (id: string) =>
   234→    setSelFiles((s) => {
   235→      const n = new Set(s ?? (activeFile ? [activeFile.id] : []));
   236→      if (n.has(id)) {
   237→        if (n.size > 1) n.delete(id);
   238→      } else {
   239→        n.add(id);
   240→      }
   241→      return n;
   242→    });
   243→
   244→  const expectedName = useMemo(() => {
   245→    if (!tool || !activeFile) return null;
   246→    const t = TOOLS_BY_ID[tool];
   247→    const target = t.accepts.includes(activeFile.kind)
   248→      ? activeFile
   249→      : files.find((f) => effectiveSel.has(f.id) && t.accepts.includes(f.kind));
   250→    return target ? resultName(tool, target.name) : null;
   251→  }, [tool, activeFile, files, effectiveSel]);
   252→
   253→  const execute = async () => {
   254→    if (!tool || running) return;
   255→    const t = TOOLS_BY_ID[tool];
   256→    const chosen = files.filter((f) => effectiveSel.has(f.id) && t.accepts.includes(f.kind));
   257→    if (chosen.length === 0) {
   258→      toast.error(t.accepts.includes("image") ? "حدد صورًا من قائمة الملفات أولًا" : "حدد ملف PDF من قائمة الملفات أولًا");
   259→      return;
   260→    }
   261→    await runOnFiles(tool, params, chosen.map((f) => f.id));
   262→    const st = useWorkspace.getState();
   263→    if (st.runState === "error") toast.error(st.runMessage);
   264→  };
   265→
   266→  const download = (r: ResultFile) => {
   267→    const embedded = downloadBlob(r.blob, r.name);
   268→    if (embedded) {
   269→      toast("بدأ التنزيل", {
   270→        description: "إن لم يظهر الملف فافتح النتيجة في تبويب مستقل",
   271→        action: { label: "فتح في تبويب", onClick: () => openBlobInNewTab(r.blob) },
   272→        duration: 10_000,
   273→      });
   274→    } else {
   275→      toast.success("بدأ التنزيل");
   276→    }
   277→  };
   278→
   279→  const commitRename = () => {
   280→    if (!renaming) return;
   281→    const name = renaming.value.trim();
   282→    if (name) renameResult(renaming.id, name);
   283→    setRenaming(null);
   284→  };
   285→
   286→  return (
   287→    <div
   288→      className="mx-auto max-w-7xl px-4 py-5"
   289→      onDragOver={(e) => {
   290→        e.preventDefault();
   291→        setDragOver(true);
   292→      }}
   293→      onDragLeave={() => setDragOver(false)}
   294→      onDrop={(e) => {
   295→        e.preventDefault();
   296→        setDragOver(false);
   297→        const fs = Array.from(e.dataTransfer.files);
   298→        if (fs.length) addFiles(fs);
   299→      }}
   300→    >
   301→      <input
   302→        ref={inputRef}
   303→        type="file"
   304→        multiple
   305→        accept=".pdf,image/*"
   306→        className="hidden"
   307→        onChange={(e) => {
   308→          const fs = Array.from(e.target.files ?? []);
   309→          if (fs.length) addFiles(fs);
   310→          e.target.value = "";
   311→        }}
   312→      />
   313→
   314→      {files.length === 0 ? (
   315→        /* ─── لا ملفات: منطقة إفلات واحدة ─── */
   316→        <div
   317→          className={cn(
   318→            "mt-8 rounded-3xl border-2 border-dashed bg-white/70 py-20 text-center transition",
   319→            dragOver ? "border-primary bg-primary/5" : "border-primary/30"

   321→          onClick={() => inputRef.current?.click()}

   326→          <div className="mx-auto grid size-16 place-items-center rounded-2xl bg-primary text-primary-foreground shadow-lg shadow-primary/25">
   327→            <Plus className="size-7" aria-hidden />
   328→          </div>
   329→          <h2 className="mt-4 text-xl font-extrabold">اسحب ملفاتك هنا</h2>
