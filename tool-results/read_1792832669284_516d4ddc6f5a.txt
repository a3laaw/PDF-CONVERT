   152→/* ─────────── الرسم على سياق كانفس (الأحجام نسبية للارتفاع h) ─────────── */
   153→
   154→function drawAnno(
   155→  ctx: CanvasRenderingContext2D,
   156→  a: Anno,
   157→  w: number,
   158→  h: number,
   159→  fontFamily: string
   160→) {
   161→  ctx.save();
   162→  ctx.lineCap = "round";
   163→  ctx.lineJoin = "round";
   164→  const pm = h / 1000; // وحدة الألف من الارتفاع → بكسل
   165→  if (a.tool === "pen" || a.tool === "highlight") {
   166→    const s = a as StrokeAnno;
   167→    if (s.pts.length === 0) {
   168→      ctx.restore();
   169→      return;
   170→    }
   171→    ctx.strokeStyle = a.tool === "highlight" ? HIGHLIGHT_COLOR : a.color;
   172→    ctx.globalAlpha = a.tool === "highlight" ? 0.42 : 1;
   173→    ctx.lineWidth = a.tool === "highlight" ? a.width * 4 * pm : a.width * pm;
   174→    ctx.beginPath();
   175→    ctx.moveTo(s.pts[0][0] * w, s.pts[0][1] * h);
   176→    for (let i = 1; i < s.pts.length; i++)
   177→      ctx.lineTo(s.pts[i][0] * w, s.pts[i][1] * h);
   178→    if (s.pts.length === 1) ctx.lineTo(s.pts[0][0] * w + pm, s.pts[0][1] * h);
   179→    ctx.stroke();
   180→  } else if (a.tool === "line" || a.tool === "arrow") {
   181→    const s = a as ShapeAnno;
   182→    ctx.strokeStyle = a.color;
   183→    ctx.lineWidth = a.width * pm;
   184→    ctx.beginPath();
   185→    ctx.moveTo(s.x1 * w, s.y1 * h);
   186→    ctx.lineTo(s.x2 * w, s.y2 * h);
   187→    ctx.stroke();
   188→    if (a.tool === "arrow") {
   189→      const ang = Math.atan2(s.y2 - s.y1, s.x2 - s.x1);
   190→      const head = Math.max(8, a.width * 4) * pm;
   191→      for (const off of [2.6, -2.6]) {
   192→        ctx.beginPath();
   193→        ctx.moveTo(s.x2 * w, s.y2 * h);
   194→        ctx.lineTo(s.x2 * w - head * Math.cos(ang - off), s.y2 * h - head * Math.sin(ang - off));
   195→        ctx.stroke();
   196→      }
   197→    }
   198→  } else if (a.tool === "rect" || a.tool === "ellipse") {
   199→    const s = a as ShapeAnno;
   200→    ctx.strokeStyle = a.color;
   201→    ctx.lineWidth = a.width * pm;
   202→    const x = Math.min(s.x1, s.x2) * w;
   203→    const y = Math.min(s.y1, s.y2) * h;
   204→    const bw = Math.abs(s.x2 - s.x1) * w;
   205→    const bh = Math.abs(s.y2 - s.y1) * h;
   206→    ctx.beginPath();
   207→    if (a.tool === "rect") ctx.rect(x, y, bw, bh);
   208→    else ctx.ellipse(x + bw / 2, y + bh / 2, bw / 2, bh / 2, 0, 0, Math.PI * 2);
   209→    ctx.stroke();
   210→  } else {
   211→    const s = a as TextAnno;
   212→    ctx.fillStyle = a.color;
   213→    ctx.direction = "rtl";
   214→    ctx.textAlign = "right";
   215→    ctx.textBaseline = "alphabetic";
   216→    ctx.font = `700 ${s.size * pm}px ${fontFamily}`;
   217→    ctx.fillText(s.text, s.x * w, s.y * h);
   218→  }
   219→  ctx.restore();
   220→}

   224→interface Box {
   225→  x0: number;
   226→  y0: number;
   227→  x1: number;
   228→  y1: number;
   229→}
   230→
   231→function annoBox(a: Anno, w: number, h: number, fam: string): Box {
   232→  const pad = (a.width * (a.tool === "highlight" ? 4 : 1)) / 2000 + 0.005;
   233→  if (a.tool === "pen" || a.tool === "highlight") {
   234→    const xs = a.pts.map((p) => p[0]);
   235→    const ys = a.pts.map((p) => p[1]);
   236→    return {
   237→      x0: Math.min(...xs) - pad,
   238→      y0: Math.min(...ys) - pad,
   239→      x1: Math.max(...xs) + pad,
   240→      y1: Math.max(...ys) + pad,
   241→    };
   242→  }
   243→  if (a.tool === "text") {
   244→    const c = document.createElement("canvas");
   245→    const ctx = c.getContext("2d")!;
   246→    ctx.font = `700 ${a.size}px ${fam}`;
   247→    const twNorm = (ctx.measureText(a.text).width * h) / 1000 / Math.max(1, w);
   248→    const asc = a.size / 1000;
   249→    return {
   250→      x0: a.x - twNorm - pad,
   251→      y0: a.y - asc * 1.05 - pad,
   252→      x1: a.x + pad,
   253→      y1: a.y + asc * 0.35 + pad,
   254→    };
   255→  }
   256→  const s = a as ShapeAnno;
   257→  return {
   258→    x0: Math.min(s.x1, s.x2) - pad,
   259→    y0: Math.min(s.y1, s.y2) - pad,
   260→    x1: Math.max(s.x1, s.x2) + pad,
   261→    y1: Math.max(s.y1, s.y2) + pad,
   262→  };
   263→}
   264→
   265→function moveAnno(a: Anno, dx: number, dy: number): Anno {
   266→  if (a.tool === "pen" || a.tool === "highlight")
   267→    return { ...a, pts: a.pts.map(([px, py]) => [px + dx, py + dy]) };
   268→  if (a.tool === "text") return { ...a, x: a.x + dx, y: a.y + dy };
   269→  const s = a as ShapeAnno;
   270→  return { ...s, x1: s.x1 + dx, y1: s.y1 + dy, x2: s.x2 + dx, y2: s.y2 + dy };
   271→}

   295→export function EditorCanvas({ fileName, kind, file, page, pages, onPage, onClose, onSave }: Props) {
   296→  const wrapRef = useRef<HTMLDivElement>(null); // منطقة الكانفس القابلة للتمرير
   297→  const baseRef = useRef<HTMLCanvasElement | null>(null); // طبقة الصفحة (عرض فقط)
   298→  const liveRef = useRef<HTMLCanvasElement>(null); // طبقة الرسم التفاعلية
   299→  const [ready, setReady] = useState(false);
   300→  const [aspect, setAspect] = useState<number | null>(null);
   301→  const [tool, setTool] = useState<ToolId>("pen");
   302→  const [color, setColor] = useState(COLORS[0].v);
   303→  const [size, setSize] = useState(3.5);
   304→  const [textSize, setTextSize] = useState(36);
   305→  const [doc, setDoc] = useState<EditorDoc>({ annos: {}, past: [], future: [] });
   306→  const [busy, setBusy] = useState<string>("");
   307→  const [typing, setTypingState] = useState<{ id: number; x: number; y: number; value: string } | null>(null);
   308→  const typingRef = useRef<{ id: number; x: number; y: number; value: string } | null>(null);
   309→  const typingSeq = useRef(0);
   310→  const setTyping = (t: { id: number; x: number; y: number; value: string } | null) => {
   311→    typingRef.current = t;
   312→    setTypingState(t);
   313→  };
   314→  const [selectedId, setSelectedId] = useState<string | null>(null);
   315→  const [zoom, setZoom] = useState(1);
   316→  const [areaW, setAreaW] = useState(760);
   317→  const drawing = useRef<Anno | null>(null);
   318→  const drag = useRef<{ id: string; orig: Anno; sx: number; sy: number } | null>(null);
   319→  const curList = doc.annos[page] ?? [];
   320→
   321→  /* عرض المنطقة المتاحة → عرض الصفحة الأساسي */
   322→  useEffect(() => {
   323→    const el = wrapRef.current;
   324→    if (!el) return;
   325→    const ro = new ResizeObserver(() => setAreaW(Math.max(260, el.clientWidth - 48)));
   326→    ro.observe(el);
   327→    setAreaW(Math.max(260, el.clientWidth - 48));
   328→    return () => ro.disconnect();
   329→  }, [ready]);
   330→
   331→  /* ── تحميل الصفحة الأساسية بدقة عالية ── */
   332→  useEffect(() => {
   333→    let alive = true;
   334→    setReady(false);
   335→    setSelectedId(null);
   336→    (async () => {
   337→      const maxCssW = 900;
   338→      if (kind === "image") {
   339→        const url = URL.createObjectURL(file);
   340→        const img = new Image();
   341→        img.src = url;
   342→        await img.decode().catch(() => undefined);
   343→        if (!alive) return;
   344→        const scale = Math.min(2, (maxCssW * 2) / img.naturalWidth);
   345→        const c = document.createElement("canvas");
   346→        c.width = Math.round(img.naturalWidth * scale);
   347→        c.height = Math.round(img.naturalHeight * scale);
   348→        const ctx = c.getContext("2d")!;
   349→        ctx.fillStyle = "#ffffff";
   350→        ctx.fillRect(0, 0, c.width, c.height);
   351→        ctx.drawImage(img, 0, 0, c.width, c.height);
   352→        URL.revokeObjectURL(url);
   353→        baseRef.current = c;
   354→        setAspect(c.width / c.height);
   355→      } else {
   356→        const pdfjs = await import("pdfjs-dist");
   357→        pdfjs.GlobalWorkerOptions.workerSrc = "/pdf.worker.min.mjs";
   358→        const bytes = new Uint8Array(await file.arrayBuffer());
   359→        const pdoc = await pdfjs.getDocument({ data: bytes }).promise;
   360→        const p = await pdoc.getPage(Math.min(Math.max(1, page), pdoc.numPages));
   361→        const v1 = p.getViewport({ scale: 1 });
   362→        const targetW = Math.max(1000, Math.min(1800, Math.round(maxCssW * 2)));
   363→        const viewport = p.getViewport({ scale: targetW / v1.width });
   364→        const c = document.createElement("canvas");
   365→        c.width = Math.floor(viewport.width);
   366→        c.height = Math.floor(viewport.height);
   367→        const ctx = c.getContext("2d")!;
   368→        ctx.fillStyle = "#ffffff";
   369→        ctx.fillRect(0, 0, c.width, c.height);
   370→        await p.render({ canvasContext: ctx, viewport } as never).promise;
   371→        if (!alive) return;
   372→        baseRef.current = c;
   373→        setAspect(c.width / c.height);
   374→        (pdoc as unknown as { destroy?: () => void }).destroy?.();
   375→      }
   376→      if (alive) setReady(true);
   377→    })().catch(() => setReady(true));
   378→    return () => {
   379→      alive = false;
   380→    };
   381→  }, [file, kind, page]);
   382→
   383→  /* انتظار جاهزية الخط ثم إعادة رسم (لضمان قياس نص عربي صحيح) */
   384→  useEffect(() => {
   385→    if (typeof document !== "undefined" && "fonts" in document)
   386→      document.fonts.ready.then(() => setZoom((z) => z)).catch(() => undefined);
   387→  }, []);
   388→
   389→  /* ── إعادة رسم الطبقة التفاعلية ── */
   390→  const repaint = useCallback(() => {
   391→    const live = liveRef.current;
   392→    const base = baseRef.current;
   393→    if (!live || !base) return;
   394→    live.width = base.width;
   395→    live.height = base.height;
   396→    const ctx = live.getContext("2d")!;
   397→    ctx.clearRect(0, 0, live.width, live.height);
   398→    const fam = getFam();
   399→    for (const a of doc.annos[page] ?? []) drawAnno(ctx, a, live.width, live.height, fam);
   400→    if (drawing.current) drawAnno(ctx, drawing.current, live.width, live.height, fam);
   401→  }, [doc.annos, page]);
   402→
   403→  useEffect(() => {
   404→    repaint();
   405→  }, [repaint, ready]);
   406→
   407→  /* ── التاريخ (ذرّي — آمن مع StrictMode) ── */
   408→  const commit = (next: Anno[]) =>
   409→    setDoc((d) => ({
   410→      annos: { ...d.annos, [page]: next },
   411→      past: [...d.past.slice(-40), d.annos],
   412→      future: [],
   413→    }));
   414→  const undo = () => {
   415→    setSelectedId(null);
   416→    setDoc((d) =>
   417→      d.past.length
   418→        ? {
   419→            annos: d.past[d.past.length - 1],
   420→            past: d.past.slice(0, -1),
   421→            future: [d.annos, ...d.future].slice(0, 40),
   422→          }
   423→        : d
   424→    );
   425→  };
   426→  const redo = () => {
   427→    setSelectedId(null);
   428→    setDoc((d) =>
   429→      d.future.length
   430→        ? { annos: d.future[0], past: [...d.past, d.annos], future: d.future.slice(1) }
   431→        : d
   432→    );
   433→  };
   434→
   435→  /* ── المؤشر → إحداثيات طبيعية ── */
   436→  const norm = (e: React.PointerEvent): [number, number] => {
   437→    const r = (e.currentTarget as HTMLCanvasElement).getBoundingClientRect();
   438→    return [
   439→      Math.min(1, Math.max(0, (e.clientX - r.left) / r.width)),
   440→      Math.min(1, Math.max(0, (e.clientY - r.top) / r.height)),
   441→    ];
   442→  };
   443→
   444→  const onDown = (e: React.PointerEvent<HTMLCanvasElement>) => {
   445→    if (typing) return;
   446→    const [x, y] = norm(e);
   447→    if (tool === "select") {
   448→      const live = liveRef.current;
   449→      if (!live) return;
   450→      let hit: Anno | null = null;
   451→      const list = doc.annos[page] ?? [];
   452→      const fam = getFam();
   453→      for (let i = list.length - 1; i >= 0; i--) {
   454→        const b = annoBox(list[i], live.width, live.height, fam);
   455→        if (boxHit(b, x, y)) {
   456→          hit = list[i];
   457→          break;
   458→        }
   459→      }
   460→      setSelectedId(hit ? hit.id : null);
   461→      if (hit) {
   462→        drag.current = { id: hit.id, orig: hit, sx: x, sy: y };
   463→        /* نقفل التاريخ قبل أول تحرك — الحركة كلها خطوة تراجع واحدة */
   464→        setDoc((d) => ({ ...d, past: [...d.past.slice(-40), d.annos], future: [] }));
   465→        e.currentTarget.setPointerCapture(e.pointerId);
   466→      }
   467→      return;
   468→    }
