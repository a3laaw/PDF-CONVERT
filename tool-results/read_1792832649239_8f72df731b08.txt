   115→async function renderPageDataUrl(file: File, pageNo: number, maxW = 820): Promise<string> {
   116→  const pdfjs = await import("pdfjs-dist");
   117→  pdfjs.GlobalWorkerOptions.workerSrc = "/pdf.worker.min.mjs";
   118→  const bytes = new Uint8Array(await file.arrayBuffer());
   119→  const doc = await pdfjs.getDocument({ data: bytes }).promise;
   120→  const page = await doc.getPage(Math.max(1, pageNo));
   121→  const v1 = page.getViewport({ scale: 1 });
   122→  const viewport = page.getViewport({ scale: Math.min(1.8, maxW / v1.width) });
   123→  const canvas = document.createElement("canvas");
   124→  canvas.width = Math.floor(viewport.width);
   125→  canvas.height = Math.floor(viewport.height);
   126→  const c2 = canvas.getContext("2d")!;
   127→  c2.fillStyle = "#ffffff";
   128→  c2.fillRect(0, 0, canvas.width, canvas.height);
   129→  await page.render({ canvas, canvasContext: c2, viewport } as never).promise;
   130→  return canvas.toDataURL("image/jpeg", 0.85);
   131→}

   133→function fmtSize(b: number): string {
   134→  if (b >= 1048576) return `${toHindiDigits((b / 1048576).toFixed(1))} م.ب`;
   135→  return `${toHindiDigits(Math.max(1, Math.round(b / 1024)))} ك.ب`;
   136→}

   138→function extLabel(name: string): string {
   139→  const e = name.match(/\.([^.]+)$/)?.[1]?.toUpperCase() ?? "ملف";
   140→  return e === "JPG" ? "JPG" : e;
   141→}

   155→  const clearResults = useWorkspace((s) => s.clearResults);
   156→  const downloadAll = useWorkspace((s) => s.downloadAll);

   158→
   159→  const [activeId, setActiveId] = useState<string | null>(null);
   160→  const [selFiles, setSelFiles] = useState<Set<string> | null>(null);
   161→  const [tool, setTool] = useState<OpId | null>(null);
   162→  const [params, setParams] = useState<Record<string, string>>({});
   163→  const [selPages, setSelPages] = useState<number[]>([]);
   164→  const [thumbs, setThumbs] = useState<Record<string, string[]>>({});
   165→  const [page, setPage] = useState(1);
   166→
   167→  const [renaming, setRenaming] = useState<{ id: string; value: string } | null>(null);
   168→  const [dragOver, setDragOver] = useState(false);
   169→  const [editing, setEditing] = useState(false);
   170→  const inputRef = useRef<HTMLInputElement>(null);
   171→  const thumbsReq = useRef<Set<string>>(new Set());
   172→
   173→  /* الملف النشط: المعيّن بالنقر أو آخر ملف أُضيف — اشتقاق بلا effects */
   174→  const activeFile = files.find((f) => f.id === activeId) ?? files[files.length - 1] ?? null;
   175→  const effectiveSel = selFiles ?? (activeFile ? new Set([activeFile.id]) : new Set<string>());
   176→  const running = runState === "running";
   177→
   178→  /* مصغرات صفحات الـ PDF النشط */
   179→  useEffect(() => {
   180→    if (!activeFile || activeFile.kind !== "pdf") return;
   181→    if (thumbs[activeFile.id] || thumbsReq.current.has(activeFile.id)) return;
   182→    thumbsReq.current.add(activeFile.id);
   183→    pdfPageThumbs(activeFile.file, 60)
   184→      .then((arr) => setThumbs((t) => ({ ...t, [activeFile.id]: arr })))
   185→      .catch(() => thumbsReq.current.delete(activeFile.id));
   186→  }, [activeFile, thumbs]);
   187→
   188→  /* المعاينة الكبيرة للصفحة النشطة — مفتوحة بالمفتاح لتفادي setState المتزامن */
   189→  const previewKey = activeFile ? `${activeFile.id}:${activeFile.kind === "image" ? "img" : page}` : "";
   190→  const [previewState, setPreviewState] = useState<{ key: string; url: string } | null>(null);
   191→  useEffect(() => {
   192→    if (!activeFile || !previewKey) return;
   193→    let alive = true;
   194→    let objUrl: string | null = null;
   195→    const done = (url: string) => {
   196→      if (alive) setPreviewState({ key: previewKey, url });
   197→    };
   198→    if (activeFile.kind === "image") {
   199→      Promise.resolve().then(() => {
   200→        objUrl = URL.createObjectURL(activeFile.file);
   201→        done(objUrl);
   202→      });
   203→    } else {
   204→      renderPageDataUrl(activeFile.file, page)
   205→        .then(done)
   206→        .catch(() => undefined);
   207→    }
   208→    return () => {
   209→      alive = false;
   210→      if (objUrl) URL.revokeObjectURL(objUrl);
   211→    };
   212→  }, [activeFile, page, previewKey]);
   213→  const preview = previewState && previewState.key === previewKey ? previewState.url : "";
   214→
   215→  const editorEnabled =
   216→    activeFile?.kind === "pdf" || activeFile?.kind === "image";
   217→
   218→  /* وضع الفصل البصري بعمودين */
   219→  const splitMode = tool === "split" && activeFile?.kind === "pdf";
   220→
   221→  const pickTool = (id: OpId) => {
   222→    setTool(id);
   223→    const def: Record<string, string> = {};
   224→    for (const f of TOOLS_BY_ID[id].fields) if (f.default !== undefined) def[f.key] = f.default;
   225→    const pf = TOOLS_BY_ID[id].fields.find((f) => f.pagesPicker);
   226→    if (pf) def[pf.key] = selPages.join(",");
   227→    setParams(def);
   228→  };
   229→
   230→  const togglePage = (n: number) => {
   231→    const next = selPages.includes(n)
   232→      ? selPages.filter((x) => x !== n)
   233→      : [...selPages, n].sort((a, b) => a - b);
   234→    setSelPages(next);
   235→    if (tool) {
   236→      const pf = TOOLS_BY_ID[tool].fields.find((f) => f.pagesPicker);
   237→      if (pf) setParams((p) => ({ ...p, [pf.key]: next.join(",") }));
   238→    }
   239→  };
   240→
   241→  const toggleSel = (id: string) =>
   242→    setSelFiles((s) => {
   243→      const n = new Set(s ?? (activeFile ? [activeFile.id] : []));
   244→      if (n.has(id)) {
   245→        if (n.size > 1) n.delete(id);
   246→      } else {
   247→        n.add(id);
   248→      }
   249→      return n;

   252→  const expectedName = useMemo(() => {
   253→    if (!tool || !activeFile) return null;
   254→    const t = TOOLS_BY_ID[tool];
   255→    const target = t.accepts.includes(activeFile.kind)
   256→      ? activeFile
   257→      : files.find((f) => effectiveSel.has(f.id) && t.accepts.includes(f.kind));
   258→    return target ? resultName(tool, target.name) : null;
   259→  }, [tool, activeFile, files, effectiveSel]);
   260→
   261→  /* ذكاء الأدوات: كل أداة تعمل فقط لو المحدد فيهم نوع تقبله */
   262→  const selKinds = useMemo(() => {
   263→    const ks = new Set<FileKind>();
   264→    for (const f of files) if (effectiveSel.has(f.id)) ks.add(f.kind);
   265→    return ks;
   266→  }, [files, effectiveSel]);

   268→  const toolEnabled = (id: OpId): boolean => {
   269→    const t = TOOLS_BY_ID[id];
   270→    if (!t) return false;
   271→    for (const k of selKinds) if (t.accepts.includes(k)) return true;

   278→    if (t.accepts.includes("image") && !t.accepts.includes("pdf"))
   279→      toast.info("الأداة دي للصور — حدّد صورة من قائمة الملفات");
   280→    else if (activeFile?.kind === "image")
   281→      toast.info("الأداة دي لملفات PDF — للصور استخدم «من صور»");
   282→    else if (activeFile && activeFile.kind !== "pdf")
   283→      toast.info("ندعم ملفات PDF والصور فقط حاليًا — أنواع أخرى غير مدعومة بعد");
   284→    else toast.info("حدّد ملف PDF من قائمة الملفات أولًا");
   285→  };
   286→
   287→  const editorSave = (name: string, blob: Blob) => {
   288→    addEditorResult(name, blob);
   289→    setEditing(false);
   290→    toast.success("حُفظت التعديلات — الملف جاهز للتنزيل");
   291→  };
   292→
   293→  const execute = async () => {

   295→    const t = TOOLS_BY_ID[tool];
