    11→let workerReady = false;
    12→async function getPdfjs() {
    13→  const pdfjs = await import("pdfjs-dist");
    14→  if (!workerReady) {
    15→    pdfjs.GlobalWorkerOptions.workerSrc = "/pdf.worker.min.mjs";
    16→    workerReady = true;
    17→  }
    18→  return pdfjs;
    19→}

    21→async function loadPdfLib(bytes: Uint8Array): Promise<PDFDocument> {
    22→  return PDFDocument.load(bytes, { ignoreEncryption: true, updateMetadata: false });
    23→}

    25→/** نص عربي → PNG شفاف عبر Canvas (تشكيل حروف مثالي بواسطة المتصفح) */
    26→async function arabicTextPng(
    27→  text: string,
    28→  opts: { size: number; color: string; bold?: boolean }
    29→): Promise<{ bytes: Uint8Array; w: number; h: number }> {
    30→  const font = `${opts.bold ? "700" : "500"} ${opts.size}px Tajawal, "Noto Sans Arabic", sans-serif`;
    31→  try {
    32→    await document.fonts.load(font, text);
    33→  } catch {
    34→    /* الخط غير متاح — نكمل بخط النظام */
    35→  }
    36→  const canvas = document.createElement("canvas");
    37→  const ctx = canvas.getContext("2d")!;
    38→  ctx.font = font;
    39→  const metrics = ctx.measureText(text);
    40→  const pad = Math.ceil(opts.size * 0.25);
    41→  canvas.width = Math.ceil(metrics.width) + pad * 2;
    42→  canvas.height = Math.ceil(opts.size * 1.45);
    43→  const ctx2 = canvas.getContext("2d")!;
    44→  ctx2.clearRect(0, 0, canvas.width, canvas.height);
    45→  ctx2.font = font;
    46→  ctx2.textAlign = "center";
    47→  ctx2.textBaseline = "middle";
    48→  ctx2.fillStyle = opts.color;
    49→  ctx2.direction = "rtl";
    50→  ctx2.fillText(text, canvas.width / 2, canvas.height / 2);
    51→  const dataUrl = canvas.toDataURL("image/png");
    52→  const res = await fetch(dataUrl);
    53→  const buf = new Uint8Array(await res.arrayBuffer());
    54→  return { bytes: buf, w: canvas.width, h: canvas.height };
    55→}

    57→function resultFile(name: string, bytes: Uint8Array | Blob, fromStep: string, kind: ResultFile["kind"]): ResultFile {
    58→  const blob = bytes instanceof Blob ? bytes : new Blob([bytes.buffer as ArrayBuffer], { type: mimeOf(name) });
    59→  return {
    60→    id: `${Date.now()}-${Math.random().toString(36).slice(2, 7)}`,
    61→    name,
    62→    blob,
    63→    url: URL.createObjectURL(blob),
    64→    size: blob.size,
    65→    fromStep,
    66→    kind,
    67→  };
    68→}
    69→
    70→function mimeOf(name: string): string {
    71→  if (name.endsWith(".pdf")) return "application/pdf";
    72→  if (name.endsWith(".png")) return "image/png";
    73→  if (name.endsWith(".jpg") || name.endsWith(".jpeg")) return "image/jpeg";
    74→  if (name.endsWith(".txt")) return "text/plain;charset=utf-8";
    75→  if (name.endsWith(".zip")) return "application/zip";
    76→  return "application/octet-stream";
    77→}

    97→export async function runOp(
    98→  op: string,
    99→  inputs: EngineInput[],
   100→  params: Record<string, string>,
   101→  label: string,
   102→  ctx: RunContext = {}
   103→): Promise<ResultFile[]> {
   104→  if (inputs.length === 0) {
   105→    throw new Error("لا توجد ملفات مناسبة لهذه الخطوة — أضف ملفاتك أولًا");
   106→  }
   107→  switch (op) {
   108→    case "merge":
   109→      return opMerge(inputs, label, ctx);
   110→    case "compress":
   111→      return opCompress(inputs, params, label, ctx);
   112→    case "extract":
   113→      return opExtractOrDelete(inputs, params, "extract", label, ctx);
   114→    case "delete":
   115→      return opExtractOrDelete(inputs, params, "delete", label, ctx);
   116→    case "rotate":
   117→      return opRotate(inputs, params, label, ctx);
   118→    case "reorder":
   119→      return opReorder(inputs, params, label, ctx);
   120→    case "split":
   121→      return opSplit(inputs, label, ctx);
   122→    case "watermark":
   123→      return opWatermark(inputs, params, label, ctx);
   124→    case "page_numbers":
   125→      return opPageNumbers(inputs, params, label, ctx);
   126→    case "images_to_pdf":
   127→      return opImagesToPdf(inputs, params, label, ctx);
   128→    case "pdf_to_images":
   129→      return opPdfToImages(inputs, params, label, ctx);
   130→    case "extract_text":
   131→      return opExtractText(inputs, label, ctx);
   132→    case "metadata":
   133→      return opMetadata(inputs, params, label, ctx);
   134→    case "nup":
   135→      return opNup(inputs, params, label, ctx);
   136→    default:
   137→      throw new Error("عملية غير معروفة");
   138→  }
   139→}

   141→/* دمج */
   142→async function opMerge(inputs: EngineInput[], label: string, ctx: RunContext): Promise<ResultFile[]> {
   143→  ctx.onProgress?.(`جارٍ دمج ${inputs.length} ملفات…`);
   144→  const out = await PDFDocument.create();
   145→  for (const input of inputs) {
   146→    const src = await loadPdfLib(await bytesOf(input));
   147→    const pages = await out.copyPages(src, src.getPageIndices());
   148→    pages.forEach((p) => out.addPage(p));
   149→  }
   150→  const bytes = await out.save({ useObjectStreams: true });
   151→  return [resultFile("الملف-المدموج.pdf", bytes, label, "pdf")];
   152→}
   153→
   154→/* تصغير الحجم — ثلاثة مستويات بذكاء:
   155→   ملف نصي → ضغط الهيكل فقط (الترقيم الصوري يكبّر النصي)
   156→   ملف ممسوح → إعادة بناء الصفحات كصور JPEG (توفير حقيقي) */
   157→async function hasTextLayer(bytes: Uint8Array): Promise<boolean> {
   158→  try {
   159→    const pdfjs = await getPdfjs();
   160→    const doc = await pdfjs.getDocument({ data: bytes.slice() }).promise;
   161→    let len = 0;
   162→    for (let i = 1; i <= Math.min(2, doc.numPages); i++) {
   163→      const page = await doc.getPage(i);
   164→      const tc = await page.getTextContent();
   165→      len += tc.items.reduce((acc, it) => acc + ("str" in it ? it.str.trim().length : 0), 0);
   166→    }
   167→    return len >= 20;
   168→  } catch {
   169→    return false;
   170→  }
   171→}

   173→async function opCompress(
   174→  inputs: EngineInput[],
   175→  params: Record<string, string>,
   176→  label: string,
   177→  ctx: RunContext
   178→): Promise<ResultFile[]> {
   179→  const level = params.level ?? "متوسط";
   180→  const results: ResultFile[] = [];
   181→  for (const input of inputs) {
   182→    ctx.onProgress?.(`جارٍ تصغير حجم «${input.name}»…`);
   183→    const bytes = await bytesOf(input);
   184→    const textual = await hasTextLayer(bytes);
   185→    if (!textual) {
   186→      // ملف ممسوح: إعادة بناء بجودة حسب المستوى
   187→      const pdfjs = await getPdfjs();
   188→      const src = await pdfjs.getDocument({ data: bytes.slice() }).promise;
   189→      const out = await PDFDocument.create();
   190→      const scale = level === "متوسط" ? 1.5 : 1.05;
   191→      const quality = level === "متوسط" ? 0.8 : 0.55;
   192→      for (let i = 1; i <= src.numPages; i++) {
   193→        ctx.onProgress?.(`إعادة بناء الصفحة ${i} من ${src.numPages}…`);
   194→        const page = await src.getPage(i);
   195→        const viewport = page.getViewport({ scale });
   196→        const canvas = document.createElement("canvas");
   197→        canvas.width = Math.floor(viewport.width);
   198→        canvas.height = Math.floor(viewport.height);
   199→        const c2 = canvas.getContext("2d")!;
   200→        c2.fillStyle = "#ffffff";
   201→        c2.fillRect(0, 0, canvas.width, canvas.height);
   202→        await page.render({ canvas, canvasContext: c2, viewport } as never).promise;
   203→        const jpegUrl = canvas.toDataURL("image/jpeg", quality);
   204→        const res = await fetch(jpegUrl);
   205→        const jpegBytes = new Uint8Array(await res.arrayBuffer());
   206→        const img = await out.embedJpg(jpegBytes);
   207→        const p = out.addPage([
   208→          page.getViewport({ scale: 1 }).width,
   209→          page.getViewport({ scale: 1 }).height,
   210→        ]);
   211→        p.drawImage(img, { x: 0, y: 0, width: p.getWidth(), height: p.getHeight() });
   212→      }
   213→      const saved = await out.save({ useObjectStreams: true });
   214→      results.push(resultFile("بعد-تصغير-الحجم.pdf", saved, label, "pdf"));
   215→    } else {
   216→      // ملف نصي: ضغط الهيكل + تنظيف البيانات في المتوسط والقوي
   217→      const doc = await loadPdfLib(bytes);
   218→      if (level !== "خفيف") {
   219→        doc.setTitle("");
   220→        doc.setAuthor("");
   221→        doc.setSubject("");
   222→        doc.setKeywords([]);
   223→      }
   224→      const saved = await doc.save({ useObjectStreams: true });
   225→      results.push(resultFile("بعد-تصغير-الحجم.pdf", saved, label, "pdf"));
   226→    }
   227→  }
   228→  return results;
   229→}

   231→/* استخراج / حذف صفحات */
   232→async function opExtractOrDelete(
   233→  inputs: EngineInput[],
   234→  params: Record<string, string>,
   235→  mode: "extract" | "delete",
   236→  label: string,
   237→  ctx: RunContext
   238→): Promise<ResultFile[]> {
   239→  const results: ResultFile[] = [];
   240→  for (const input of inputs) {
   241→    ctx.onProgress?.(`جارٍ ${mode === "extract" ? "استخراج" : "حذف"} الصفحات من «${input.name}»…`);
   242→    const src = await loadPdfLib(await bytesOf(input));
   243→    const total = src.getPageCount();
   244→    const pages = parsePageRanges(params.pages ?? "", total);
   245→    const keep =
   246→      mode === "extract"
   247→        ? pages.map((p) => p - 1)
   248→        : Array.from({ length: total }, (_, i) => i + 1)
   249→            .filter((p) => !pages.includes(p))
   250→            .map((p) => p - 1);
   251→    if (keep.length === 0) {
   252→      throw new Error("لا توجد صفحات متبقية بعد هذه العملية — راجع الأرقام المختارة");
   253→    }
   254→    const out = await PDFDocument.create();
   255→    const copied = await out.copyPages(src, keep);
   256→    copied.forEach((p) => out.addPage(p));
   257→    const saved = await out.save({ useObjectStreams: true });
   258→    results.push(
   259→      resultFile(mode === "extract" ? "الصفحات-المستخرجة.pdf" : "بعد-حذف-الصفحات.pdf", saved, label, "pdf")
   260→    );
   261→  }
   262→  return results;
   263→}

   265→/* تدوير */
   266→async function opRotate(
   267→  inputs: EngineInput[],
   268→  params: Record<string, string>,
   269→  label: string,
   270→  ctx: RunContext
   271→): Promise<ResultFile[]> {
   272→  const angle = Number(params.angle ?? "90");
   273→  const results: ResultFile[] = [];
   274→  for (const input of inputs) {
   275→    ctx.onProgress?.(`جارٍ تدوير «${input.name}»…`);
   276→    const doc = await loadPdfLib(await bytesOf(input));
   277→    const total = doc.getPageCount();
   278→    const pages = params.pages ? parsePageRanges(params.pages, total) : null;
   279→    doc.getPages().forEach((page, idx) => {
   280→      if (pages && !pages.includes(idx + 1)) return;
   281→      const current = page.getRotation().angle;
   282→      page.setRotation(degrees((current + angle) % 360));
   283→    });
   284→    const saved = await doc.save({ useObjectStreams: true });
   285→    results.push(resultFile("بعد-التدوير.pdf", saved, label, "pdf"));
   286→  }
   287→  return results;
   288→}

   290→/* ترتيب */
   291→async function opReorder(
   292→  inputs: EngineInput[],
   293→  params: Record<string, string>,
   294→  label: string,
   295→  ctx: RunContext
   296→): Promise<ResultFile[]> {
   297→  const results: ResultFile[] = [];
   298→  for (const input of inputs) {
   299→    ctx.onProgress?.(`جارٍ إعادة الترتيب…`);
   300→    const src = await loadPdfLib(await bytesOf(input));
   301→    const total = src.getPageCount();
   302→    let order: number[];
   303→    if (params.mode === "مخصص" && params.order) {
   304→      order = parsePageRanges(params.order, total);
   305→    } else {
   306→      order = Array.from({ length: total }, (_, i) => total - i);
   307→    }
   308→    const out = await PDFDocument.create();
   309→    const copied = await out.copyPages(src, order.map((p) => p - 1));
   310→    copied.forEach((p) => out.addPage(p));
   311→    const saved = await out.save({ useObjectStreams: true });
   312→    results.push(resultFile("بعد-الترتيب.pdf", saved, label, "pdf"));
   313→  }
   314→  return results;
   315→}

   317→/* فصل كل صفحة */
   318→async function opSplit(inputs: EngineInput[], label: string, ctx: RunContext): Promise<ResultFile[]> {
   319→  const results: ResultFile[] = [];
   320→  const prefix = inputs.length > 1; // عدة ملفات معًا → نسبق الأسماء ب اسم الملف لتفادي التكرار
   321→  for (const input of inputs) {
   322→    ctx.onProgress?.(`جارٍ فصل صفحات «${input.name}»…`);
   323→    const src = await loadPdfLib(await bytesOf(input));
   324→    const total = src.getPageCount();
   325→    const base = prefix ? input.name.replace(/\.pdf$/i, "") + "-" : "";
   326→    for (let i = 0; i < total; i++) {
   327→      ctx.onProgress?.(`فصل «${input.name}»: الصفحة ${toHindiDigits(i + 1)} من ${toHindiDigits(total)}`);
   328→      const out = await PDFDocument.create();
   329→      const [page] = await out.copyPages(src, [i]);
   330→      out.addPage(page);
   331→      const saved = await out.save({ useObjectStreams: true });
   332→      results.push(resultFile(`${base}صفحة-${toHindiDigits(i + 1)}.pdf`, saved, label, "pdf"));
   333→    }
   334→  }
   335→  return results;
   336→}

   338→/* ختم (علامة مائية) */
   339→async function opWatermark(
   340→  inputs: EngineInput[],
   341→  params: Record<string, string>,
   342→  label: string,
   343→  ctx: RunContext
   344→): Promise<ResultFile[]> {
   345→  const text = params.text || "مسودة";
   346→  const colorMap: Record<string, string> = { رمادي: "#6b7280", احمر: "#dc2626", ازرق: "#2563eb" };
   347→  const color = colorMap[params.color ?? "رمادي"] ?? "#6b7280";
   348→  const { bytes: png, w, h } = await arabicTextPng(text, { size: 88, color, bold: true });
   349→  const results: ResultFile[] = [];
   350→  for (const input of inputs) {
   351→    ctx.onProgress?.(`جارٍ ختم «${input.name}»…`);
   352→    const doc = await loadPdfLib(await bytesOf(input));
   353→    const img = await doc.embedPng(png);
   354→    for (const page of doc.getPages()) {
   355→      const pw = page.getWidth();
   356→      const ph = page.getHeight();
   357→      const targetW = pw * 0.62;
   358→      const targetH = (h / w) * targetW;
   359→      const theta = (45 * Math.PI) / 180;
   360→      const cx = pw / 2;
   361→      const cy = ph / 2;
   362→      const ox = (targetW / 2) * Math.cos(theta) - (targetH / 2) * Math.sin(theta);
   363→      const oy = (targetW / 2) * Math.sin(theta) + (targetH / 2) * Math.cos(theta);
   364→      page.drawImage(img, {
   365→        x: cx - ox,
   366→        y: cy - oy,
   367→        width: targetW,
   368→        height: targetH,
   369→        rotate: degrees(45),
   370→        opacity: 0.16,
   371→      });
   372→    }
   373→    const saved = await doc.save({ useObjectStreams: true });
   374→    results.push(resultFile("بعد-الختم.pdf", saved, label, "pdf"));
   375→  }
   376→  return results;
   377→}

   379→/* ترقيم الصفحات */
   380→async function opPageNumbers(
   381→  inputs: EngineInput[],
   382→  params: Record<string, string>,
   383→  label: string,
   384→  ctx: RunContext
   385→): Promise<ResultFile[]> {
   386→  const results: ResultFile[] = [];
   387→  for (const input of inputs) {
   388→    ctx.onProgress?.(`جارٍ ترقيم «${input.name}»…`);
   389→    const doc = await loadPdfLib(await bytesOf(input));
   390→    const total = doc.getPageCount();
   391→    const hindi = (params.digits ?? "هندي") === "هندي";
   392→    const fmt = params.format ?? "رقم فقط";
   393→    for (let i = 1; i <= total; i++) {
   394→      const shown = hindi ? toHindiDigits(i) : String(i);
   395→      const text = fmt === "صفحة من" ? `صفحة ${shown} من ${hindi ? toHindiDigits(total) : total}` : shown;
   396→      const { bytes: png, w, h } = await arabicTextPng(text, { size: 60, color: "#1b2438" });
   397→      const img = await doc.embedPng(png);
   398→      const page = doc.getPage(i - 1);
   399→      const targetH = 22;
   400→      const targetW = (w / h) * targetH;
   401→      page.drawImage(img, {
   402→        x: (page.getWidth() - targetW) / 2,
   403→        y: 16,
   404→        width: targetW,
   405→        height: targetH,
   406→        opacity: 0.85,
   407→      });
   408→    }
   409→    const saved = await doc.save({ useObjectStreams: true });
   410→    results.push(resultFile("بعد-الترقيم.pdf", saved, label, "pdf"));
   411→  }
   412→  return results;
   413→}

   415→/* صور إلى PDF */
   416→async function opImagesToPdf(
   417→  inputs: EngineInput[],
   418→  params: Record<string, string>,
   419→  label: string,
   420→  ctx: RunContext
   421→): Promise<ResultFile[]> {
   422→  ctx.onProgress?.(`جارٍ تحويل ${inputs.length} صور إلى PDF…`);
   423→  const out = await PDFDocument.create();
   424→  for (const input of inputs) {
   425→    const bytes = await bytesOf(input);
   426→    const name = input.name.toLowerCase();
   427→    let img;
   428→    if (name.endsWith(".png")) img = await out.embedPng(bytes);
   429→    else img = await out.embedJpg(bytes);
   430→    if ((params.layout ?? "ملائم تلقائي") === "A4") {
   431→      const page = out.addPage([595.28, 841.89]);
   432→      const margin = 28;
   433→      const maxW = page.getWidth() - margin * 2;
   434→      const maxH = page.getHeight() - margin * 2;
   435→      const scale = Math.min(maxW / img.width, maxH / img.height);
   436→      page.drawImage(img, {
   437→        x: (page.getWidth() - img.width * scale) / 2,
   438→        y: (page.getHeight() - img.height * scale) / 2,
   439→        width: img.width * scale,
   440→        height: img.height * scale,
   441→      });
   442→    } else {
   443→      const page = out.addPage([img.width, img.height]);
   444→      page.drawImage(img, { x: 0, y: 0, width: img.width, height: img.height });
   445→    }
   446→  }
   447→  const saved = await out.save({ useObjectStreams: true });
   448→  return [resultFile("من-الصور.pdf", saved, label, "pdf")];
   449→}

   451→/* PDF إلى صور */
   452→async function opPdfToImages(
   453→  inputs: EngineInput[],
   454→  params: Record<string, string>,
   455→  label: string,
   456→  ctx: RunContext
   457→): Promise<ResultFile[]> {
   458→  const pdfjs = await getPdfjs();
   459→  const format = params.format ?? "png";
   460→  const scaleMap: Record<string, number> = { مضغوطة: 1.2, متوازنة: 2.0, عالية: 3.0 };
   461→  const scale = scaleMap[params.quality ?? "متوازنة"] ?? 2.0;
   462→  const results: ResultFile[] = [];
   463→  for (const input of inputs) {
   464→    const bytes = await bytesOf(input);
   465→    const src = await pdfjs.getDocument({ data: bytes.slice() }).promise;
   466→    for (let i = 1; i <= src.numPages; i++) {
   467→      ctx.onProgress?.(`تحويل الصفحة ${i} من ${src.numPages} إلى صورة…`);
   468→      const page = await src.getPage(i);
   469→      const viewport = page.getViewport({ scale });
   470→      const canvas = document.createElement("canvas");
   471→      canvas.width = Math.floor(viewport.width);
   472→      canvas.height = Math.floor(viewport.height);
   473→      const c2 = canvas.getContext("2d")!;
   474→      c2.fillStyle = "#ffffff";
   475→      c2.fillRect(0, 0, canvas.width, canvas.height);
   476→      await page.render({ canvas, canvasContext: c2, viewport } as never).promise;
   477→      const mime = format === "jpeg" ? "image/jpeg" : "image/png";
   478→      const blob = await new Promise<Blob>((resolve, reject) =>
   479→        canvas.toBlob((b) => (b ? resolve(b) : reject(new Error("تعذر إنشاء الصورة"))), mime, 0.9)
   480→      );
   481→      const ext = format === "jpeg" ? "jpg" : "png";
   482→      results.push(resultFile(`${input.name.replace(/\.pdf$/i, "")}-صفحة-${i}.${ext}`, blob, label, "image"));
   483→    }
   484→  }
   485→  return results;
   486→}

   488→/* استخراج النص */
   489→async function opExtractText(inputs: EngineInput[], label: string, ctx: RunContext): Promise<ResultFile[]> {
   490→  const pdfjs = await getPdfjs();
   491→  const results: ResultFile[] = [];
   492→  for (const input of inputs) {
   493→    ctx.onProgress?.(`جارٍ استخراج نصوص «${input.name}»…`);
   494→    const bytes = await bytesOf(input);
   495→    const src = await pdfjs.getDocument({ data: bytes.slice() }).promise;
   496→    const chunks: string[] = [];
   497→    for (let i = 1; i <= src.numPages; i++) {
   498→      const page = await src.getPage(i);
   499→      const content = await page.getTextContent();
   500→      const line = content.items
   501→        .map((it) => ("str" in it ? it.str : ""))
   502→        .join(" ")
   503→        .replace(/\s+/g, " ")
   504→        .trim();
   505→      chunks.push(`── الصفحة ${toHindiDigits(i)} ──\n${line}`);
   506→    }
   507→    const text = chunks.join("\n\n");
   508→    const blob = new Blob(["\ufeff" + text], { type: "text/plain;charset=utf-8" });
   509→    results.push(resultFile("نص-الملف.txt", blob, label, "other"));
   510→  }
   511→  return results;
   512→}

   535→/* عدة صفحات في الورقة */
   536→async function opNup(
   537→  inputs: EngineInput[],
   538→  params: Record<string, string>,
   539→  label: string,
   540→  ctx: RunContext
   541→): Promise<ResultFile[]> {
   542→  const results: ResultFile[] = [];
   543→  for (const input of inputs) {
   544→    ctx.onProgress?.(`جارٍ تجميع الصفحات في أوراق…`);
   545→    const src = await loadPdfLib(await bytesOf(input));
   546→    const out = await PDFDocument.create();
   547→    const indices = src.getPageIndices();
   548→    for (let i = 0; i < indices.length; i += 2) {
   549→      const pair = indices.slice(i, i + 2);
   550→      const embedded = await out.embedPdf(src, pair);
   551→      const page = out.addPage([841.89, 595.28]); // A4 عرضي
   552→      const slotW = page.getWidth() / 2 - 18;
   553→      const slotH = page.getHeight() - 24;
   554→      embedded.forEach((emb, j) => {
   555→        const scale = Math.min(slotW / emb.width, slotH / emb.height);
   556→        const w = emb.width * scale;
   557→        const h = emb.height * scale;
   558→        const x = j === 0 ? page.getWidth() - 12 - w : 12; // RTL: الأولى يمين
   559→        page.drawPage(emb, { x, y: (page.getHeight() - h) / 2, width: w, height: h });
   560→      });
   561→    }
   562→    const saved = await out.save({ useObjectStreams: true });
   563→    results.push(resultFile("صفحتان-في-الورقة.pdf", saved, label, "pdf"));
   564→  }
   565→  return results;
   566→}

   576→/** فحص PDF سريع: عدد الصفحات + مصغّرة + هل هو ممسوح ضوئيًا */
   577→export async function inspectPdf(file: File): Promise<InspectInfo> {
   578→  const pdfjs = await getPdfjs();
   579→  const bytes = new Uint8Array(await file.arrayBuffer());
   580→  const doc = await pdfjs.getDocument({ data: bytes.slice() }).promise;
   581→  const pages = doc.numPages;
   582→
   583→  // مصغّرة الصفحة الأولى
   584→  const page = await doc.getPage(1);
   585→  const viewport = page.getViewport({ scale: 0.4 });
   586→  const canvas = document.createElement("canvas");
   587→  canvas.width = Math.floor(viewport.width);
   588→  canvas.height = Math.floor(viewport.height);
   589→  const c2 = canvas.getContext("2d")!;
   590→  c2.fillStyle = "#ffffff";
   591→  c2.fillRect(0, 0, canvas.width, canvas.height);
   592→  await page.render({ canvas, canvasContext: c2, viewport } as never).promise;
   593→  const thumb = canvas.toDataURL("image/png");
   594→
   595→  // كشف الممسوح: نص الصفحتين الأوليين
   596→  let textLen = 0;
   597→  for (let i = 1; i <= Math.min(2, pages); i++) {
   598→    const p = await doc.getPage(i);
   599→    const tc = await p.getTextContent();
   600→    textLen += tc.items.reduce((acc, it) => acc + ("str" in it ? it.str.trim().length : 0), 0);
   601→  }
   602→  return { pages, thumb, scanned: textLen < 20 };
   603→}

   605→/** مصغّرة صورة */
   606→export async function inspectImage(file: File): Promise<{ thumb: string }> {
   607→  return new Promise((resolve) => {
   608→    const url = URL.createObjectURL(file);
   609→    const img = new Image();
   610→    img.onload = () => {
   611→      const canvas = document.createElement("canvas");
   612→      const scale = Math.min(1, 220 / img.width);
   613→      canvas.width = Math.floor(img.width * scale);
   614→      canvas.height = Math.floor(img.height * scale);
   615→      canvas.getContext("2d")!.drawImage(img, 0, 0, canvas.width, canvas.height);
   616→      URL.revokeObjectURL(url);
   617→      resolve({ thumb: canvas.toDataURL("image/png") });
   618→    };
   619→    img.onerror = () => {
   620→      URL.revokeObjectURL(url);
   621→      resolve({ thumb: "" });
   622→    };
   623→    img.src = url;
   624→  });
   625→}

   649→/** إنشاء ZIP من النتائج */
   650→export async function zipResults(files: ResultFile[]): Promise<Blob> {
   651→  const JSZip = (await import("jszip")).default;
   652→  const zip = new JSZip();
   653→  const used = new Set<string>();
   654→  for (const f of files) {
   655→    let name = f.name;
   656→    let k = 2;
   657→    while (used.has(name)) {
   658→      name = f.name.replace(/(\.[^.]+)$/, `-${k}$1`);
   659→      k++;
   660→    }
   661→    used.add(name);
   662→    zip.file(name, f.blob);
   663→  }
   664→  return zip.generateAsync({ type: "blob", compression: "DEFLATE" });
   665→}

   667→/** تنزيل ملف */
   668→export function downloadBlob(blob: Blob, name: string): boolean {
   669→  const url = URL.createObjectURL(blob);
   670→  const a = document.createElement("a");
   671→  a.href = url;
   672→  a.download = name;
   673→  a.rel = "noopener";
   674→  document.body.appendChild(a);
   675→  a.click();
   676→  a.remove();
   677→  // مهلة أطول حتى يبقى الرابط صالحًا إن احتاج المستخدم فتحه من تبويب مستقل
   678→  setTimeout(() => URL.revokeObjectURL(url), 60_000);
   679→  return isEmbedded();
   680→}

   692→export function openBlobInNewTab(blob: Blob): void {
   693→  const url = URL.createObjectURL(blob);
   694→  const w = window.open(url, "_blank");
   695→  if (!w) {
   696→    // حاجب النوافذ منع الفتح — نترك الرابط حيًا مدة أطول ونجرب عبر رابط مخفي
   697→    const a = document.createElement("a");

   700→    a.rel = "noopener";
   701→    document.body.appendChild(a);
   702→    a.click();
   703→    a.remove();
   704→  }
   705→  setTimeout(() => URL.revokeObjectURL(url), 120_000);
   706→}
