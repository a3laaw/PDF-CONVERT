   118→  5333→@app.post("/api/run")
   119→  5334→async def run(
   120→  5335→    files: list[UploadFile] = File(...),
   121→  5336→    steps: str = Form(...),
   122→  5337→    ref_names: str = Form(""),
   123→  5338→):
   124→  5339→    """
   125→  5340→    تنفيذ خط أنابيب كامل في طلب واحد (بدون حالة).
   126→  5341→    steps: [{"op":"compress","inputs":["@f0"],"params":{}}]
   127→  5342→      مراجع المدخلات: @f0/@f1 = ترتيب الملفات المرفوعة | @prev = مخرج السابق
   128→  5343→      | @all_pdfs / @all_images / @all_cad
   129→  5344→    النتيجة: ZIP يحتوي كل الملفات الناتجة + manifest.json
   130→  5345→    """
   131→  5346→    try:
   132→  5347→        step_list = json.loads(steps)
   133→  5348→    except Exception:
   134→  5349→        raise HTTPException(400, "steps غير صالح")
   135→  5350→
   136→  5351→    with tempfile.TemporaryDirectory() as td:
   137→  5352→        tdp = Path(td)
   138→  5353→        in_dir, out_dir = tdp / "in", tdp / "out"
   139→  5354→        in_dir.mkdir(); out_dir.mkdir()
   140→  5355→
   141→  5356→        uploaded: list[Path] = []
   142→  5357→        for f in files:
   143→  5358→            uploaded.append(_save_upload(f, in_dir))
   144→  5359→
   145→  5360→        def resolve(refs):
   146→  5361→            paths = []
   147→  5362→            for r in refs or []:
   148→  5363→                if r == "@prev":
   149→  5364→                    if not step_outputs:
   150→  5365→                        raise HTTPException(400, "لا توجد خطوة سابقة")
   151→  5366→                    paths.append(list(step_outputs.values())[-1])
   152→  5367→                elif r.startswith("@f") and r[2:].isdigit():
   153→  5368→                    idx = int(r[2:])
   154→  5369→                    if idx >= len(uploaded):
   155→  5370→                        raise HTTPException(400, f"مرجع ملف غير موجود: {r}")
   156→  5371→                    paths.append(uploaded[idx])
   157→  5372→                elif r == "@all_pdfs":
   158→  5373→                    paths.extend(p for p in uploaded if p.suffix.lower() == ".pdf")
   159→  5374→                elif r == "@all_images":
   160→  5375→                    paths.extend(p for p in uploaded if p.suffix.lower() in (".png", ".jpg", ".jpeg", ".gif", ".webp", ".bmp"))
   161→  5376→                elif r == "@all_cad":
   162→  5377→                    paths.extend(p for p in uploaded if p.suffix.lower() in (".dxf", ".dwg", ".rvt"))
   163→  5378→                elif r == "@all":
   164→  5379→                    paths.extend(uploaded)
   165→  5380→                else:
   166→  5381→                    raise HTTPException(400, f"مرجع غير معروف: {r}")
   167→  5382→            return paths
   168→  5383→
   169→  5384→        step_outputs: dict[str, Path] = {}
   170→  5385→        results = []
   171→  5386→        t0 = time.time()
   172→  5387→
   173→  5388→        for i, step in enumerate(step_list):
   174→  5389→            op = step.get("op", "")
   175→  5390→            params = step.get("params", {}) or {}
   176→  5391→            paths = resolve(step.get("inputs") or ["@all"])
   177→  5392→            if not paths:
   178→  5393→                raise HTTPException(400, f"الخطوة {i+1}: لا توجد مدخلات")
   179→  5394→            t1 = time.time()
   180→  5395→            try:
   181→  5396→                outs = run_op(op, paths, params, out_dir)
   182→  5397→            except OpError as e:
   183→  5398→                raise HTTPException(400, f"الخطوة {i+1} ({op}): {e}")
   184→  5399→            except Exception as e:
   185→  5400→                raise HTTPException(500, f"الخطوة {i+1} ({op}): {type(e).__name__}: {e}")
   186→  5401→            step_outputs[f"step{i}"] = outs[-1]
   187→  5402→            results.append({
   188→  5403→                "step": i + 1, "op": op,
   189→  5404→                "label": step.get("label", op),
   190→  5405→                "duration": round(time.time() - t1, 2),
   191→  5406→                "outputs": [{"name": o.name, "size": o.stat().st_size} for o in outs],
   192→  5407→            })
   193→  5408→
   194→  5409→        # حزم كل المخرجات في ZIP
   195→  5410→        buf = io.BytesIO()
   196→  5411→        with zipfile.ZipFile(buf, "w", zipfile.ZIP_DEFLATED) as z:
   197→  5412→            for o in sorted(out_dir.rglob("*")):
   198→  5413→                if o.is_file():
   199→  5414→                    z.write(o, o.relative_to(out_dir))
   200→  5415→            z.writestr("manifest.json", json.dumps({
   201→  5416→                "ok": True,
   202→  5417→                "total_duration": round(time.time() - t0, 2),
   203→  5418→                "results": results,
   204→  5419→            }, ensure_ascii=False, indent=2))
   205→  5420→        buf.seek(0)
   206→  5421→        return Response(
   207→  5422→            content=buf.read(),
   208→  5423→            media_type="application/zip",
   209→  5424→            headers={"Content-Disposition": "attachment; filename=pdf-station-results.zip"},
   210→  5425→        )
