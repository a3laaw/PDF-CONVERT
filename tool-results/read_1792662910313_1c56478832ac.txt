    12→let workerReady = false;
    13→async function getPdfjs() {
    14→  const pdfjs = await import("pdfjs-dist");
    15→  if (!workerReady) {
    16→    pdfjs.GlobalWorkerOptions.workerSrc = "/pdf.worker.min.mjs";
    17→    workerReady = true;
    18→  }
    19→  return pdfjs;
    20→}

    22→async function loadPdfLib(bytes: Uint8Array): Promise<PDFDocument> {
    23→  return PDFDocument.load(bytes, { ignoreEncryption: true, updateMetadata: false });
    24→}

    26→/** نص عربي → PNG شفاف عبر Canvas (تشكيل حروف مثالي بواسطة المتصفح) */
    27→async function arabicTextPng(
    28→  text: string,
    29→  opts: { size: number; color: string; bold?: boolean }
    30→): Promise<{ bytes: Uint8Array; w: number; h: number }> {
    31→  const font = `${opts.bold ? "700" : "500"} ${opts.size}px Tajawal, "Noto Sans Arabic", sans-serif`;
    32→  try {
    33→    await document.fonts.load(font, text);
    34→  } catch {
    35→    /* الخط غير متاح — نكمل بخط النظام */
    36→  }
    37→  const canvas = document.createElement("canvas");
    38→  const ctx = canvas.getContext("2d")!;
    39→  ctx.font = font;
    40→  const metrics = ctx.measureText(text);
    41→  const pad = Math.ceil(opts.size * 0.25);
    42→  canvas.width = Math.ceil(metrics.width) + pad * 2;
    43→  canvas.height = Math.ceil(opts.size * 1.45);
    44→  const ctx2 = canvas.getContext("2d")!;
    45→  ctx2.clearRect(0, 0, canvas.width, canvas.height);
    46→  ctx2.font = font;
    47→  ctx2.textAlign = "center";
    48→  ctx2.textBaseline = "middle";
    49→  ctx2.fillStyle = opts.color;
    50→  ctx2.direction = "rtl";
    51→  ctx2.fillText(text, canvas.width / 2, canvas.height / 2);
    52→  const dataUrl = canvas.toDataURL("image/png");
    53→  const res = await fetch(dataUrl);
    54→  const buf = new Uint8Array(await res.arrayBuffer());
    55→  return { bytes: buf, w: canvas.width, h: canvas.height };
    56→}

    58→function resultFile(name: string, bytes: Uint8Array | Blob, fromStep: string, kind: ResultFile["kind"]): ResultFile {
    59→  const blob = bytes instanceof Blob ? bytes : new Blob([bytes.buffer as ArrayBuffer], { type: mimeOf(name) });
    60→  return {
    61→    id: `${Date.now()}-${Math.random().toString(36).slice(2, 7)}`,
    62→    name,
    63→    blob,
    64→    url: URL.createObjectURL(blob),
    65→    size: blob.size,
    66→    fromStep,
    67→    kind,
    68→  };
    69→}
    70→
    71→function mimeOf(name: string): string {
    72→  if (name.endsWith(".pdf")) return "application/pdf";
    73→  if (name.endsWith(".png")) return "image/png";
    74→  if (name.endsWith(".jpg") || name.endsWith(".jpeg")) return "image/jpeg";
    75→  if (name.endsWith(".txt")) return "text/plain;charset=utf-8";
    76→  if (name.endsWith(".zip")) return "application/zip";
    77→  return "application/octet-stream";
    78→}

    98→export async function runOp(
    99→  op: string,
   100→  inputs: EngineInput[],
   101→  params: Record<string, string>,
   102→  label: string,
   103→  ctx: RunContext = {}
   104→): Promise<ResultFile[]> {
   105→  if (inputs.length === 0) {
   106→    throw new Error("لا توجد ملفات مناسبة لهذه الخطوة — أضف ملفاتك أولًا");
   107→  }
   108→  switch (op) {
   109→    case "merge":
   110→      return opMerge(inputs, label, ctx);
   111→    case "compress":
   112→      return opCompress(inputs, params, label, ctx);
   113→    case "extract":
   114→      return opExtractOrDelete(inputs, params, "extract", label, ctx);
   115→    case "delete":
   116→      return opExtractOrDelete(inputs, params, "delete", label, ctx);
   117→    case "rotate":
   118→      return opRotate(inputs, params, label, ctx);
   119→    case "reorder":
   120→      return opReorder(inputs, params, label, ctx);
   121→    case "split":
   122→      return opSplit(inputs, label, ctx);
   123→    case "watermark":
   124→      return opWatermark(inputs, params, label, ctx);
   125→    case "page_numbers":
   126→      return opPageNumbers(inputs, params, label, ctx);
   127→    case "images_to_pdf":
   128→      return opImagesToPdf(inputs, params, label, ctx);
   129→    case "pdf_to_images":
   130→      return opPdfToImages(inputs, params, label, ctx);
   131→    case "extract_text":
   132→      return opExtractText(inputs, label, ctx);
   133→    case "metadata":
   134→      return opMetadata(inputs, params, label, ctx);
   135→    case "nup":
   136→      return opNup(inputs, params, label, ctx);
   137→    default:
   138→      throw new Error("عملية غير معروفة");
   139→  }
   140→}

   142→/* دمج */
   143→async function opMerge(inputs: EngineInput[], label: string, ctx: RunContext): Promise<ResultFile[]> {
   144→  ctx.onProgress?.(`جارٍ دمج ${inputs.length} ملفات…`);
   145→  const out = await PDFDocument.create();
   146→  for (const input of inputs) {
   147→    const src = await loadPdfLib(await bytesOf(input));
   148→    const pages = await out.copyPages(src, src.getPageIndices());
   149→    pages.forEach((p) => out.addPage(p));
   150→  }
   151→  const bytes = await out.save({ useObjectStreams: true });
   152→  return [resultFile(resultName("merge", inputs[0].name), bytes, label, "pdf")];
   153→}
   154→
   155→/* تصغير الحجم — ثلاثة مستويات بذكاء:
   156→   ملف نصي → ضغط الهيكل فقط (الترقيم الصوري يكبّر النصي)
   157→   ملف ممسوح → إعادة بناء الصفحات كصور JPEG (توفير حقيقي) */
   158→async function hasTextLayer(bytes: Uint8Array): Promise<boolean> {
   159→  try {
   160→    const pdfjs = await getPdfjs();
   161→    const doc = await pdfjs.getDocument({ data: bytes.slice() }).promise;
   162→    let len = 0;
   163→    for (let i = 1; i <= Math.min(2, doc.numPages); i++) {
   164→      const page = await doc.getPage(i);
   165→      const tc = await page.getTextContent();
   166→      len += tc.items.reduce((acc, it) => acc + ("str" in it ? it.str.trim().length : 0), 0);
   167→    }
   168→    return len >= 20;
   169→  } catch {
   170→    return false;
   171→  }
   172→}

   174→async function opCompress(
   175→  inputs: EngineInput[],
   176→  params: Record<string, string>,
   177→  label: string,
   178→  ctx: RunContext
   179→): Promise<ResultFile[]> {
   180→  const level = params.level ?? "متوسط";
   181→  const results: ResultFile[] = [];
   182→  for (const input of inputs) {
   183→    ctx.onProgress?.(`جارٍ تصغير حجم «${input.name}»…`);
   184→    const bytes = await bytesOf(input);
   185→    const textual = await hasTextLayer(bytes);
   186→    if (!textual) {
   187→      // ملف ممسوح: إعادة بناء بجودة حسب المستوى
   188→      const pdfjs = await getPdfjs();
   189→      const src = await pdfjs.getDocument({ data: bytes.slice() }).promise;
   190→      const out = await PDFDocument.create();
   191→      const scale = level === "متوسط" ? 1.5 : 1.05;
   192→      const quality = level === "متوسط" ? 0.8 : 0.55;
   193→      for (let i = 1; i <= src.numPages; i++) {
   194→        ctx.onProgress?.(`إعادة بناء الصفحة ${i} من ${src.numPages}…`);
   195→        const page = await src.getPage(i);
   196→        const viewport = page.getViewport({ scale });
   197→        const canvas = document.createElement("canvas");
   198→        canvas.width = Math.floor(viewport.width);
   199→        canvas.height = Math.floor(viewport.height);
   200→        const c2 = canvas.getContext("2d")!;
   201→        c2.fillStyle = "#ffffff";
   202→        c2.fillRect(0, 0, canvas.width, canvas.height);
   203→        await page.render({ canvas, canvasContext: c2, viewport } as never).promise;
   204→        const jpegUrl = canvas.toDataURL("image/jpeg", quality);
   205→        const res = await fetch(jpegUrl);
   206→        const jpegBytes = new Uint8Array(await res.arrayBuffer());
   207→        const img = await out.embedJpg(jpegBytes);
   208→        const p = out.addPage([
   209→          page.getViewport({ scale: 1 }).width,
   210→          page.getViewport({ scale: 1 }).height,
   211→        ]);
   212→        p.drawImage(img, { x: 0, y: 0, width: p.getWidth(), height: p.getHeight() });
   213→      }
   214→      const saved = await out.save({ useObjectStreams: true });
   215→      results.push(resultFile(resultName("compress", input.name), saved, label, "pdf"));
   216→    } else {
   217→      // ملف نصي: ضغط الهيكل + تنظيف البيانات في المتوسط والقوي
   218→      const doc = await loadPdfLib(bytes);
   219→      if (level !== "خفيف") {
   220→        doc.setTitle("");
   221→        doc.setAuthor("");
   222→        doc.setSubject("");
   223→        doc.setKeywords([]);
   224→      }
   225→      const saved = await doc.save({ useObjectStreams: true });
   226→      results.push(resultFile(resultName("compress", input.name), saved, label, "pdf"));
   227→    }
   228→  }
   229→  return results;
   230→}

   232→/* استخراج / حذف صفحات */
   233→async function opExtractOrDelete(
   234→  inputs: EngineInput[],
   235→  params: Record<string, string>,
   236→  mode: "extract" | "delete",
   237→  label: string,
   238→  ctx: RunContext
   239→): Promise<ResultFile[]> {
   240→  const results: ResultFile[] = [];
   241→  for (const input of inputs) {
   242→    ctx.onProgress?.(`جارٍ ${mode === "extract" ? "استخراج" : "حذف"} الصفحات من «${input.name}»…`);
   243→    const src = await loadPdfLib(await bytesOf(input));
   244→    const total = src.getPageCount();
   245→    const pages = parsePageRanges(params.pages ?? "", total);
   246→    const keep =
   247→      mode === "extract"
   248→        ? pages.map((p) => p - 1)
   249→        : Array.from({ length: total }, (_, i) => i + 1)
   250→            .filter((p) => !pages.includes(p))
   251→            .map((p) => p - 1);
   252→    if (keep.length === 0) {
   253→      throw new Error("لا توجد صفحات متبقية بعد هذه العملية — راجع الأرقام المختارة");
   254→    }
   255→    const out = await PDFDocument.create();
   256→    const copied = await out.copyPages(src, keep);
   257→    copied.forEach((p) => out.addPage(p));
   258→    const saved = await out.save({ useObjectStreams: true });
   259→    results.push(resultFile(resultName(mode, input.name), saved, label, "pdf"));
   260→  }
   261→  return results;
   262→}

   264→/* تدوير */
   265→async function opRotate(
   266→  inputs: EngineInput[],
   267→  params: Record<string, string>,
   268→  label: string,
   269→  ctx: RunContext
   270→): Promise<ResultFile[]> {
   271→  const angle = Number(params.angle ?? "90");
   272→  const results: ResultFile[] = [];
   273→  for (const input of inputs) {
   274→    ctx.onProgress?.(`جارٍ تدوير «${input.name}»…`);
   275→    const doc = await loadPdfLib(await bytesOf(input));
   276→    const total = doc.getPageCount();
   277→    const pages = params.pages ? parsePageRanges(params.pages, total) : null;
   278→    doc.getPages().forEach((page, idx) => {
   279→      if (pages && !pages.includes(idx + 1)) return;
   280→      const current = page.getRotation().angle;
   281→      page.setRotation(degrees((current + angle) % 360));
   282→    });
   283→    const saved = await doc.save({ useObjectStreams: true });
   284→    results.push(resultFile(resultName("rotate", input.name), saved, label, "pdf"));
   285→  }
   286→  return results;
   287→}

   289→/* ترتيب */
   290→async function opReorder(
   291→  inputs: EngineInput[],
   292→  params: Record<string, string>,
   293→  label: string,
   294→  ctx: RunContext
   295→): Promise<ResultFile[]> {
   296→  const results: ResultFile[] = [];
   297→  for (const input of inputs) {
   298→    ctx.onProgress?.(`جارٍ إعادة الترتيب…`);
   299→    const src = await loadPdfLib(await bytesOf(input));
   300→    const total = src.getPageCount();
   301→    let order: number[];
   302→    if (params.mode === "مخصص" && params.order) {
   303→      order = parsePageRanges(params.order, total);
   304→    } else {
   305→      order = Array.from({ length: total }, (_, i) => total - i);
   306→    }
   307→    const out = await PDFDocument.create();
   308→    const copied = await out.copyPages(src, order.map((p) => p - 1));
   309→    copied.forEach((p) => out.addPage(p));
   310→    const saved = await out.save({ useObjectStreams: true });
   311→    results.push(resultFile(resultName("reorder", input.name), saved, label, "pdf"));
   312→  }
   313→  return results;
   314→}

   316→/* فصل كل صفحة */
   317→async function opSplit(inputs: EngineInput[], label: string, ctx: RunContext): Promise<ResultFile[]> {
   318→  const results: ResultFile[] = [];
   319→  for (const input of inputs) {
   320→    ctx.onProgress?.(`جارٍ فصل صفحات «${input.name}»…`);
   321→    const src = await loadPdfLib(await bytesOf(input));
   322→    const total = src.getPageCount();
   323→    for (let i = 0; i < total; i++) {
   324→      ctx.onProgress?.(`فصل «${input.name}»: الصفحة ${toHindiDigits(i + 1)} من ${toHindiDigits(total)}`);
   325→      const out = await PDFDocument.create();
   326→      const [page] = await out.copyPages(src, [i]);
   327→      out.addPage(page);
   328→      const saved = await out.save({ useObjectStreams: true });
   329→      results.push(resultFile(resultName("split", input.name, i), saved, label, "pdf"));
   330→    }
   331→  }
   332→  return results;
   333→}

   335→/* ختم (علامة مائية) */
   336→async function opWatermark(
   337→  inputs: EngineInput[],
   338→  params: Record<string, string>,
   339→  label: string,
   340→  ctx: RunContext
   341→): Promise<ResultFile[]> {
   342→  const text = params.text || "مسودة";
   343→  const colorMap: Record<string, string> = { رمادي: "#6b7280", احمر: "#dc2626", ازرق: "#2563eb" };
   344→  const color = colorMap[params.color ?? "رمادي"] ?? "#6b7280";
   345→  const { bytes: png, w, h } = await arabicTextPng(text, { size: 88, color, bold: true });
   346→  const results: ResultFile[] = [];
   347→  for (const input of inputs) {
   348→    ctx.onProgress?.(`جارٍ ختم «${input.name}»…`);
   349→    const doc = await loadPdfLib(await bytesOf(input));
   350→    const img = await doc.embedPng(png);
   351→    for (const page of doc.getPages()) {
   352→      const pw = page.getWidth();
   353→      const ph = page.getHeight();
   354→      const targetW = pw * 0.62;
   355→      const targetH = (h / w) * targetW;
   356→      const theta = (45 * Math.PI) / 180;
   357→      const cx = pw / 2;
   358→      const cy = ph / 2;
   359→      const ox = (targetW / 2) * Math.cos(theta) - (targetH / 2) * Math.sin(theta);
   360→      const oy = (targetW / 2) * Math.sin(theta) + (targetH / 2) * Math.cos(theta);
   361→      page.drawImage(img, {
   362→        x: cx - ox,
   363→        y: cy - oy,
   364→        width: targetW,
   365→        height: targetH,
   366→        rotate: degrees(45),
   367→        opacity: 0.16,
   368→      });
   369→    }
   370→    const saved = await doc.save({ useObjectStreams: true });
   371→    results.push(resultFile(resultName("watermark", input.name), saved, label, "pdf"));
   372→  }
   373→  return results;
   374→}

   376→/* ترقيم الصفحات */
   377→async function opPageNumbers(
   378→  inputs: EngineInput[],
   379→  params: Record<string, string>,
   380→  label: string,
   381→  ctx: RunContext
   382→): Promise<ResultFile[]> {
   383→  const results: ResultFile[] = [];
   384→  for (const input of inputs) {
   385→    ctx.onProgress?.(`جارٍ ترقيم «${input.name}»…`);
   386→    const doc = await loadPdfLib(await bytesOf(input));
   387→    const total = doc.getPageCount();
   388→    const hindi = (params.digits ?? "هندي") === "هندي";
   389→    const fmt = params.format ?? "رقم فقط";
   390→    for (let i = 1; i <= total; i++) {
   391→      const shown = hindi ? toHindiDigits(i) : String(i);
   392→      const text = fmt === "صفحة من" ? `صفحة ${shown} من ${hindi ? toHindiDigits(total) : total}` : shown;
   393→      const { bytes: png, w, h } = await arabicTextPng(text, { size: 60, color: "#1b2438" });
   394→      const img = await doc.embedPng(png);
   395→      const page = doc.getPage(i - 1);
   396→      const targetH = 22;
   397→      const targetW = (w / h) * targetH;
   398→      page.drawImage(img, {
   399→        x: (page.getWidth() - targetW) / 2,
   400→        y: 16,
   401→        width: targetW,
   402→        height: targetH,
   403→        opacity: 0.85,
   404→      });
   405→    }
   406→    const saved = await doc.save({ useObjectStreams: true });
   407→    results.push(resultFile(resultName("page_numbers", input.name), saved, label, "pdf"));
   408→  }
   409→  return results;
   410→}

   412→/* صور إلى PDF */
   413→async function opImagesToPdf(
   414→  inputs: EngineInput[],
   415→  params: Record<string, string>,
   416→  label: string,
   417→  ctx: RunContext
   418→): Promise<ResultFile[]> {
   419→  ctx.onProgress?.(`جارٍ تحويل ${inputs.length} صور إلى PDF…`);
   420→  const out = await PDFDocument.create();
   421→  for (const input of inputs) {
   422→    const bytes = await bytesOf(input);
   423→    const name = input.name.toLowerCase();
   424→    let img;
   425→    if (name.endsWith(".png")) img = await out.embedPng(bytes);
   426→    else img = await out.embedJpg(bytes);
   427→    if ((params.layout ?? "ملائم تلقائي") === "A4") {
   428→      const page = out.addPage([595.28, 841.89]);
   429→      const margin = 28;
   430→      const maxW = page.getWidth() - margin * 2;
   431→      const maxH = page.getHeight() - margin * 2;
   432→      const scale = Math.min(maxW / img.width, maxH / img.height);
   433→      page.drawImage(img, {
   434→        x: (page.getWidth() - img.width * scale) / 2,
   435→        y: (page.getHeight() - img.height * scale) / 2,
   436→        width: img.width * scale,
   437→        height: img.height * scale,
   438→      });
   439→    } else {
   440→      const page = out.addPage([img.width, img.height]);
   441→      page.drawImage(img, { x: 0, y: 0, width: img.width, height: img.height });
   442→    }
   443→  }
   444→  const saved = await out.save({ useObjectStreams: true });
   445→  return [resultFile(resultName("images_to_pdf", inputs[0].name), saved, label, "pdf")];
   446→}

   448→/* PDF إلى صور */
   449→async function opPdfToImages(
   450→  inputs: EngineInput[],
   451→  params: Record<string, string>,
   452→  label: string,
   453→  ctx: RunContext
   454→): Promise<ResultFile[]> {
   455→  const pdfjs = await getPdfjs();
   456→  const format = params.format ?? "png";
   457→  const scaleMap: Record<string, number> = { مضغوطة: 1.2, متوازنة: 2.0, عالية: 3.0 };
   458→  const scale = scaleMap[params.quality ?? "متوازنة"] ?? 2.0;
   459→  const results: ResultFile[] = [];
   460→  for (const input of inputs) {
   461→    const bytes = await bytesOf(input);
   462→    const src = await pdfjs.getDocument({ data: bytes.slice() }).promise;
   463→    for (let i = 1; i <= src.numPages; i++) {
   464→      ctx.onProgress?.(`تحويل الصفحة ${i} من ${src.numPages} إلى صورة…`);
   465→      const page = await src.getPage(i);
   466→      const viewport = page.getViewport({ scale });
   467→      const canvas = document.createElement("canvas");
   468→      canvas.width = Math.floor(viewport.width);
   469→      canvas.height = Math.floor(viewport.height);
   470→      const c2 = canvas.getContext("2d")!;
   471→      c2.fillStyle = "#ffffff";
   472→      c2.fillRect(0, 0, canvas.width, canvas.height);
   473→      await page.render({ canvas, canvasContext: c2, viewport } as never).promise;
   474→      const mime = format === "jpeg" ? "image/jpeg" : "image/png";
   475→      const blob = await new Promise<Blob>((resolve, reject) =>
   476→        canvas.toBlob((b) => (b ? resolve(b) : reject(new Error("تعذر إنشاء الصورة"))), mime, 0.9)
   477→      );
   478→      const ext = format === "jpeg" ? "jpg" : "png";
   479→      results.push(resultFile(resultName("pdf_to_images", input.name, i - 1, "." + ext), blob, label, "image"));
   480→    }
   481→  }
   482→  return results;

   485→/* استخراج النص */
   486→async function opExtractText(inputs: EngineInput[], label: string, ctx: RunContext): Promise<ResultFile[]> {
   487→  const pdfjs = await getPdfjs();
   488→  const results: ResultFile[] = [];
   489→  for (const input of inputs) {
   490→    ctx.onProgress?.(`جارٍ استخراج نصوص «${input.name}»…`);
   491→    const bytes = await bytesOf(input);
   492→    const src = await pdfjs.getDocument({ data: bytes.slice() }).promise;
   493→    const chunks: string[] = [];
   494→    for (let i = 1; i <= src.numPages; i++) {
   495→      const page = await src.getPage(i);
   496→      const content = await page.getTextContent();
   497→      const line = content.items
   498→        .map((it) => ("str" in it ? it.str : ""))
   499→        .join(" ")
   500→        .replace(/\s+/g, " ")
   501→        .trim();
   502→      chunks.push(`── الصفحة ${toHindiDigits(i)} ──\n${line}`);
   503→    }
   504→    const text = chunks.join("\n\n");
   505→    const blob = new Blob(["\ufeff" + text], { type: "text/plain;charset=utf-8" });
   506→    results.push(resultFile(resultName("extract_text", input.name), blob, label, "other"));
   507→  }
   508→  return results;
   509→}

   532→/* عدة صفحات في الورقة */
   533→async function opNup(
   534→  inputs: EngineInput[],
   535→  params: Record<string, string>,
   536→  label: string,
   537→  ctx: RunContext
   538→): Promise<ResultFile[]> {
   539→  const results: ResultFile[] = [];
   540→  for (const input of inputs) {
   541→    ctx.onProgress?.(`جارٍ تجميع الصفحات في أوراق…`);
   542→    const src = await loadPdfLib(await bytesOf(input));
   543→    const out = await PDFDocument.create();
   544→    const indices = src.getPageIndices();
   545→    for (let i = 0; i < indices.length; i += 2) {
   546→      const pair = indices.slice(i, i + 2);
   547→      const embedded = await out.embedPdf(src, pair);
   548→      const page = out.addPage([841.89, 595.28]); // A4 عرضي
   549→      const slotW = page.getWidth() / 2 - 18;
   550→      const slotH = page.getHeight() - 24;
   551→      embedded.forEach((emb, j) => {
   552→        const scale = Math.min(slotW / emb.width, slotH / emb.height);
   553→        const w = emb.width * scale;
   554→        const h = emb.height * scale;
   555→        const x = j === 0 ? page.getWidth() - 12 - w : 12; // RTL: الأولى يمين
   556→        page.drawPage(emb, { x, y: (page.getHeight() - h) / 2, width: w, height: h });
   557→      });
   558→    }
   559→    const saved = await out.save({ useObjectStreams: true });
   560→    results.push(resultFile(resultName("nup", input.name), saved, label, "pdf"));
   561→  }
   562→  return results;
   563→}

   573→/** فحص PDF سريع: عدد الصفحات + مصغّرة + هل هو ممسوح ضوئيًا */
   574→export async function inspectPdf(file: File): Promise<InspectInfo> {
   575→  const pdfjs = await getPdfjs();
   576→  const bytes = new Uint8Array(await file.arrayBuffer());
   577→  const doc = await pdfjs.getDocument({ data: bytes.slice() }).promise;
   578→  const pages = doc.numPages;
   579→
   580→  // مصغّرة الصفحة الأولى
   581→  const page = await doc.getPage(1);
   582→  const viewport = page.getViewport({ scale: 0.4 });
   583→  const canvas = document.createElement("canvas");
   584→  canvas.width = Math.floor(viewport.width);
   585→  canvas.height = Math.floor(viewport.height);
   586→  const c2 = canvas.getContext("2d")!;
   587→  c2.fillStyle = "#ffffff";
   588→  c2.fillRect(0, 0, canvas.width, canvas.height);
   589→  await page.render({ canvas, canvasContext: c2, viewport } as never).promise;
   590→  const thumb = canvas.toDataURL("image/png");
   591→
   592→  // كشف الممسوح: نص الصفحتين الأوليين
   593→  let textLen = 0;
   594→  for (let i = 1; i <= Math.min(2, pages); i++) {
   595→    const p = await doc.getPage(i);
   596→    const tc = await p.getTextContent();
   597→    textLen += tc.items.reduce((acc, it) => acc + ("str" in it ? it.str.trim().length : 0), 0);
   598→  }
   599→  return { pages, thumb, scanned: textLen < 20 };
   600→}

   602→/** مصغّرة صورة */
   603→export async function inspectImage(file: File): Promise<{ thumb: string }> {
   604→  return new Promise((resolve) => {
   605→    const url = URL.createObjectURL(file);
   606→    const img = new Image();
   607→    img.onload = () => {
   608→      const canvas = document.createElement("canvas");
   609→      const scale = Math.min(1, 220 / img.width);
   610→      canvas.width = Math.floor(img.width * scale);
   611→      canvas.height = Math.floor(img.height * scale);
   612→      canvas.getContext("2d")!.drawImage(img, 0, 0, canvas.width, canvas.height);
   613→      URL.revokeObjectURL(url);
   614→      resolve({ thumb: canvas.toDataURL("image/png") });
   615→    };
   616→    img.onerror = () => {
   617→      URL.revokeObjectURL(url);
   618→      resolve({ thumb: "" });
   619→    };
   620→    img.src = url;
   621→  });
   622→}

   646→/** إنشاء ZIP من النتائج */
   647→export async function zipResults(files: ResultFile[]): Promise<Blob> {
   648→  const JSZip = (await import("jszip")).default;
   649→  const zip = new JSZip();
   650→  const used = new Set<string>();
   651→  for (const f of files) {
   652→    let name = f.name;
   653→    let k = 2;
   654→    while (used.has(name)) {
   655→      name = f.name.replace(/(\.[^.]+)$/, `-${k}$1`);
   656→      k++;
   657→    }
   658→    used.add(name);
   659→    zip.file(name, f.blob);
   660→  }
   661→  return zip.generateAsync({ type: "blob", compression: "DEFLATE" });
   662→}

   664→/** تنزيل ملف */
   665→export function downloadBlob(blob: Blob, name: string): boolean {
   666→  const url = URL.createObjectURL(blob);
   667→  const a = document.createElement("a");
   668→  a.href = url;
   669→  a.download = name;
   670→  a.rel = "noopener";
   671→  document.body.appendChild(a);
   672→  a.click();
   673→  a.remove();
   674→  // مهلة أطول حتى يبقى الرابط صالحًا إن احتاج المستخدم فتحه من تبويب مستقل
   675→  setTimeout(() => URL.revokeObjectURL(url), 60_000);
   676→  return isEmbedded();
   677→}

   689→export function openBlobInNewTab(blob: Blob): void {
   690→  const url = URL.createObjectURL(blob);
   691→  const w = window.open(url, "_blank");
   692→  if (!w) {
   693→    // حاجب النوافذ منع الفتح — نترك الرابط حيًا مدة أطول ونجرب عبر رابط مخفي
   694→    const a = document.createElement("a");

   696→    a.target = "_blank";
   697→    a.rel = "noopener";
   698→    document.body.appendChild(a);
   699→    a.click();
   700→    a.remove();
   701→  }
   702→  setTimeout(() => URL.revokeObjectURL(url), 120_000);
   703→}
