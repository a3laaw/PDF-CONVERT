   114→   114→async function renderPageDataUrl(file: File, pageNo: number, maxW = 820): Promise<string> {
   115→   115→  const pdfjs = await import("pdfjs-dist");
   116→   116→  pdfjs.GlobalWorkerOptions.workerSrc = "/pdf.worker.min.mjs";
   117→   117→  const bytes = new Uint8Array(await file.arrayBuffer());
   118→   118→  const doc = await pdfjs.getDocument({ data: bytes }).promise;
   119→   119→  const page = await doc.getPage(Math.max(1, pageNo));
   120→   120→  const v1 = page.getViewport({ scale: 1 });
   121→   121→  const viewport = page.getViewport({ scale: Math.min(1.8, maxW / v1.width) });
   122→   122→  const canvas = document.createElement("canvas");
   123→   123→  canvas.width = Math.floor(viewport.width);
   124→   124→  canvas.height = Math.floor(viewport.height);
   125→   125→  const c2 = canvas.getContext("2d")!;
   126→   126→  c2.fillStyle = "#ffffff";
   127→   127→  c2.fillRect(0, 0, canvas.width, canvas.height);
   128→   128→  await page.render({ canvas, canvasContext: c2, viewport } as never).promise;
   129→   129→  return canvas.toDataURL("image/jpeg", 0.85);
   130→   130→}

   132→   132→function fmtSize(b: number): string {
   133→   133→  if (b >= 1048576) return `${toHindiDigits((b / 1048576).toFixed(1).replace(".", "٫"))} م.ب`;
   134→   134→  return `${toHindiDigits(Math.max(1, Math.round(b / 1024)))} ك.ب`;
   135→   135→}

   137→   137→function extLabel(name: string): string {
   138→   138→  const e = name.match(/\.([^.]+)$/)?.[1]?.toUpperCase() ?? "ملف";
