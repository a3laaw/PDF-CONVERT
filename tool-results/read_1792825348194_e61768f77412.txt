   111→async function renderPageDataUrl(file: File, pageNo: number, maxW = 820): Promise<string> {
   112→  const pdfjs = await import("pdfjs-dist");
   113→  pdfjs.GlobalWorkerOptions.workerSrc = "/pdf.worker.min.mjs";
   114→  const bytes = new Uint8Array(await file.arrayBuffer());
   115→  const doc = await pdfjs.getDocument({ data: bytes }).promise;
   116→  const page = await doc.getPage(Math.max(1, pageNo));
   117→  const v1 = page.getViewport({ scale: 1 });
   118→  const viewport = page.getViewport({ scale: Math.min(1.8, maxW / v1.width) });
   119→  const canvas = document.createElement("canvas");
   120→  canvas.width = Math.floor(viewport.width);
   121→  canvas.height = Math.floor(viewport.height);
   122→  const c2 = canvas.getContext("2d")!;
   123→  c2.fillStyle = "#ffffff";
   124→  c2.fillRect(0, 0, canvas.width, canvas.height);
   125→  await page.render({ canvas, canvasContext: c2, viewport } as never).promise;
   126→  return canvas.toDataURL("image/jpeg", 0.85);
   127→}

   129→function fmtSize(b: number): string {
   130→  if (b >= 1048576) return `${toHindiDigits((b / 1048576).toFixed(1).replace(".", "٫"))} م.ب`;
   131→  return `${toHindiDigits(Math.max(1, Math.round(b / 1024)))} ك.ب`;
   132→}
   133→
   134→function extLabel(name: string): string {
   135→  const e = name.match(/\.([^.]+)$/)?.[1]?.toUpperCase() ?? "ملف";
   136→  return e === "JPG" ? "JPG" : e;
   137→}

   141→export function Workbench({
   142→  onOpenStudio,
   143→}: {
   144→  onOpenStudio: (v: "cad-dwg" | "cad-rvt" | "cad-adobe") => void;
   145→}) {
   146→  const files = useWorkspace((s) => s.files);
   147→  const results = useWorkspace((s) => s.results);
   148→  const runState = useWorkspace((s) => s.runState);
   149→  const runMessage = useWorkspace((s) => s.runMessage);
   150→  const runProgress = useWorkspace((s) => s.runProgress);
   151→  const addFiles = useWorkspace((s) => s.addFiles);
   152→  const removeFile = useWorkspace((s) => s.removeFile);
   153→  const runOnFiles = useWorkspace((s) => s.runOnFiles);
   154→  const renameResult = useWorkspace((s) => s.renameResult);
   155→  const clearResults = useWorkspace((s) => s.clearResults);
   156→  const downloadAll = useWorkspace((s) => s.downloadAll);
   157→
   158→  const [activeId, setActiveId] = useState<string | null>(null);
   159→  const [selFiles, setSelFiles] = useState<Set<string> | null>(null);
   160→  const [tool, setTool] = useState<OpId | null>(null);
   161→  const [params, setParams] = useState<Record<string, string>>({});
   162→  const [selPages, setSelPages] = useState<number[]>([]);
   163→  const [thumbs, setThumbs] = useState<Record<string, string[]>>({});
   164→  const [page, setPage] = useState(1);
   165→
   166→  const [renaming, setRenaming] = useState<{ id: string; value: string } | null>(null);
   167→  const [dragOver, setDragOver] = useState(false);
   168→  const inputRef = useRef<HTMLInputElement>(null);
   169→  const thumbsReq = useRef<Set<string>>(new Set());
   170→
   171→  /* الملف النشط: المعيّن بالنقر أو آخر ملف أُضيف — اشتقاق بلا effects */
   172→  const activeFile = files.find((f) => f.id === activeId) ?? files[files.length - 1] ?? null;
   173→  const effectiveSel = selFiles ?? (activeFile ? new Set([activeFile.id]) : new Set<string>());
   174→  const running = runState === "running";
   175→
   176→  /* مصغرات صفحات الـ PDF النشط */
   177→  useEffect(() => {
   178→    if (!activeFile || activeFile.kind !== "pdf") return;
   179→    if (thumbs[activeFile.id] || thumbsReq.current.has(activeFile.id)) return;
   180→    thumbsReq.current.add(activeFile.id);
   181→    pdfPageThumbs(activeFile.file, 60)
   182→      .then((arr) => setThumbs((t) => ({ ...t, [activeFile.id]: arr })))
   183→      .catch(() => thumbsReq.current.delete(activeFile.id));
   184→  }, [activeFile, thumbs]);
   185→
   186→  /* المعاينة الكبيرة للصفحة النشطة — مفتوحة بالمفتاح لتفادي setState المتزامن */
   187→  const previewKey = activeFile ? `${activeFile.id}:${activeFile.kind === "image" ? "img" : page}` : "";
   188→  const [previewState, setPreviewState] = useState<{ key: string; url: string } | null>(null);
   189→  useEffect(() => {
   190→    if (!activeFile || !previewKey) return;
   191→    let alive = true;
   192→    let objUrl: string | null = null;
   193→    const done = (url: string) => {
   194→      if (alive) setPreviewState({ key: previewKey, url });
   195→    };
   196→    if (activeFile.kind === "image") {
   197→      Promise.resolve().then(() => {
   198→        objUrl = URL.createObjectURL(activeFile.file);
   199→        done(objUrl);
   200→      });
   201→    } else {
   202→      renderPageDataUrl(activeFile.file, page)
   203→        .then(done)
   204→        .catch(() => undefined);
   205→    }
   206→    return () => {
   207→      alive = false;
   208→      if (objUrl) URL.revokeObjectURL(objUrl);
   209→    };
   210→  }, [activeFile, page, previewKey]);
   211→  const preview = previewState && previewState.key === previewKey ? previewState.url : "";
   212→
   213→  const pickTool = (id: OpId | "convert") => {
   214→    if (id === "convert") {
   215→      /* توجيه ذكي: حسب نوع الملف المحدد — نفس فكرة الاستوديوهات */
   216→      const probe =
   217→        activeFile ??
   218→        files.find((f) => effectiveSel.has(f.id)) ??
   219→        null;
   220→      const ext = probe ? (probe.name.split(".").pop() || "").toLowerCase() : "";
   221→      onOpenStudio(
   222→        ext === "rvt"
   223→          ? "cad-rvt"
   224→          : ext === "dxf" || ext === "dwg" || probe?.kind === "cad"
   225→            ? "cad-dwg"
   226→            : "cad-adobe"
   227→      );
   228→      return;
   229→    }
   230→    setTool(id);
   231→    const def: Record<string, string> = {};
   232→    for (const f of TOOLS_BY_ID[id].fields) if (f.default !== undefined) def[f.key] = f.default;
   233→    const pf = TOOLS_BY_ID[id].fields.find((f) => f.pagesPicker);
   234→    if (pf) def[pf.key] = selPages.join(",");
   235→    setParams(def);
   236→  };
   237→
   238→  const togglePage = (n: number) => {
   239→    const next = selPages.includes(n)
   240→      ? selPages.filter((x) => x !== n)
   241→      : [...selPages, n].sort((a, b) => a - b);
   242→    setSelPages(next);
   243→    if (tool) {
   244→      const pf = TOOLS_BY_ID[tool].fields.find((f) => f.pagesPicker);
   245→      if (pf) setParams((p) => ({ ...p, [pf.key]: next.join(",") }));
   246→    }
   247→  };
   248→
   249→  const toggleSel = (id: string) =>
   250→    setSelFiles((s) => {
   251→      const n = new Set(s ?? (activeFile ? [activeFile.id] : []));
   252→      if (n.has(id)) {
   253→        if (n.size > 1) n.delete(id);
   254→      } else {
   255→        n.add(id);
   256→      }
   257→      return n;
   258→    });
   259→
   260→  const expectedName = useMemo(() => {
   261→    if (!tool || !activeFile) return null;
   262→    const t = TOOLS_BY_ID[tool];
   263→    const target = t.accepts.includes(activeFile.kind)
   264→      ? activeFile
   265→      : files.find((f) => effectiveSel.has(f.id) && t.accepts.includes(f.kind));
   266→    return target ? resultName(tool, target.name) : null;
   267→  }, [tool, activeFile, files, effectiveSel]);
   268→
   269→  /* ذكاء الأدوات: كل أداة تعمل فقط لو المحدد فيهم نوع تقبله */
   270→  const selKinds = useMemo(() => {
   271→    const ks = new Set<FileKind>();
   272→    for (const f of files) if (effectiveSel.has(f.id)) ks.add(f.kind);
   273→    return ks;
   274→  }, [files, effectiveSel]);
   275→
   276→  const toolEnabled = (id: OpId): boolean => {
   277→    const t = TOOLS_BY_ID[id];
   278→    if (!t) return false;
   279→    for (const k of selKinds) if (t.accepts.includes(k)) return true;
   280→    return false;
   281→  };
   282→
   283→  const hintDisabled = (id: OpId) => {
   284→    const t = TOOLS_BY_ID[id];
   285→    if (!t) return;
   286→    if (t.accepts.includes("image") && !t.accepts.includes("pdf"))
   287→      toast.info("الأداة دي للصور — حدّد صورة من قائمة الملفات");
   288→    else if (activeFile?.kind === "image")
   289→      toast.info("الأداة دي لملفات PDF — للصور استخدم «من صور»");
   290→    else if (activeFile && activeFile.kind !== "pdf")
   291→      toast.info("النوع ده يُعرض ويتحول من استوديوهات أوتوكاد وريفيت وأدوبي في الأعلى");
   292→    else toast.info("حدّد ملف PDF من قائمة الملفات أولًا");
   293→  };
   294→
   295→  const execute = async () => {
   296→    if (!tool || running) return;
   297→    const t = TOOLS_BY_ID[tool];
   298→    const chosen = files.filter((f) => effectiveSel.has(f.id) && t.accepts.includes(f.kind));
   299→    if (chosen.length === 0) {
   300→      toast.error(t.accepts.includes("image") ? "حدد صورًا من قائمة الملفات أولًا" : "حدد ملف PDF من قائمة الملفات أولًا");
   301→      return;
   302→    }
   303→    await runOnFiles(tool, params, chosen.map((f) => f.id));
   304→    const st = useWorkspace.getState();
   305→    if (st.runState === "error") toast.error(st.runMessage);
   306→  };
   307→
   308→  const download = (r: ResultFile) => {
   309→    const embedded = downloadBlob(r.blob, r.name);
   310→    if (embedded) {
   311→      toast("بدأ التنزيل", {
   312→        description: "إن لم يظهر الملف فافتح النتيجة في تبويب مستقل",
   313→        action: { label: "فتح في تبويب", onClick: () => openBlobInNewTab(r.blob) },
   314→        duration: 10_000,
   315→      });
   316→    } else {
   317→      toast.success("بدأ التنزيل");
   318→    }
   319→  };
   320→
   321→  const commitRename = () => {
   322→    if (!renaming) return;
   323→    const name = renaming.value.trim();
   324→    if (name) renameResult(renaming.id, name);
   325→    setRenaming(null);
   326→  };
   327→
   328→  return (
   329→    <div
   330→      className="mx-auto max-w-7xl px-4 py-5"
   331→      onDragOver={(e) => {
   332→        e.preventDefault();
   333→        setDragOver(true);
   334→      }}
   335→      onDragLeave={() => setDragOver(false)}
   336→      onDrop={(e) => {
   337→        e.preventDefault();
   338→        setDragOver(false);
   339→        const fs = Array.from(e.dataTransfer.files);
   340→        if (fs.length) addFiles(fs);
   341→      }}
   342→    >
   343→      <input
   344→        ref={inputRef}
   345→        type="file"
   346→        multiple
   347→        accept=".pdf,image/*"
   348→        className="hidden"
   349→        onChange={(e) => {
   350→          const fs = Array.from(e.target.files ?? []);
   351→          if (fs.length) addFiles(fs);
   352→          e.target.value = "";
   353→        }}
   354→      />
   355→
   356→      {files.length === 0 ? (
   357→        /* ─── لا ملفات: منطقة إفلات واحدة ─── */
   358→        <div
   359→          className={cn(
   360→            "mt-8 rounded-3xl border-2 border-dashed bg-white/70 py-20 text-center transition",
   361→            dragOver ? "border-primary bg-primary/5" : "border-primary/30"
   362→          )}
   363→          onClick={() => inputRef.current?.click()}
   364→          role="button"
   365→          tabIndex={0}
   366→          onKeyDown={(e) => e.key === "Enter" && inputRef.current?.click()}
   367→        >
   368→          <div className="mx-auto grid size-16 place-items-center rounded-2xl bg-primary text-primary-foreground shadow-lg shadow-primary/25">
   369→            <Plus className="size-7" aria-hidden />
   370→          </div>
   371→          <h2 className="mt-4 text-xl font-extrabold">اسحب ملفاتك هنا</h2>
   372→          <p className="mt-1 text-sm text-muted-foreground">PDF أو صور</p>
   373→          <Button
   374→            className="mt-5 px-8 font-extrabold"
   375→            onClick={(e) => {
   376→              e.stopPropagation();
   377→              inputRef.current?.click();
   378→            }}
   379→          >
   380→            اختيار ملفات
   381→          </Button>
   382→        </div>
   383→      ) : (
   384→        <>
   385→          {/* ─── شريط الأدوات ─── */}
   386→          <div className="flex flex-wrap gap-0.5 rounded-2xl border border-border bg-white p-1.5 shadow-sm">
   387→            {TOOLBAR.map(({ id, label }) => {
   388→              const ok = id === "convert" || toolEnabled(id as OpId);
   389→              return (
   390→                <button
   391→                  key={id}
   392→                  onClick={() => (ok ? pickTool(id) : hintDisabled(id as OpId))}
   393→                  className={cn(
   394→                    "flex min-w-[62px] flex-col items-center gap-1 rounded-xl px-3 py-2 text-[11px] font-bold transition",
   395→                    tool === id
   396→                      ? "bg-primary/10 text-primary outline outline-1 outline-primary/30"
   397→                      : ok
   398→                        ? "text-muted-foreground hover:bg-primary/5 hover:text-primary"

   400→                  )}
   401→                >
   402→                  {TOOL_ICONS[id]}
   403→                  {label}
   404→                </button>
   405→              );
   406→            })}
   407→          </div>
   408→
   409→          {/* ─── الجسم: ملفات + صفحات + معاينة ─── */}
   410→          <div className="mt-4 grid gap-4 lg:grid-cols-[200px_130px_1fr]">
   411→            <div className="flex flex-col gap-2 rounded-2xl border border-border bg-white p-3">
   412→              <p className="px-1 text-[11px] font-bold text-muted-foreground">الملفات</p>
   413→              <div className="flex max-h-[460px] flex-col gap-2 overflow-y-auto">
   414→                {files.map((f) => (
   415→                  <div
   416→                    key={f.id}
   417→                    onClick={() => {
   418→                      setActiveId(f.id);
   419→                      setPage(1);
