  5333→@app.post("/api/run")
  5334→async def run(
  5335→    files: list[UploadFile] = File(...),
  5336→    steps: str = Form(...),
  5337→    ref_names: str = Form(""),
  5338→):
  5339→    """
  5340→    تنفيذ خط أنابيب كامل في طلب واحد (بدون حالة).
  5341→    steps: [{"op":"compress","inputs":["@f0"],"params":{}}]
  5342→      مراجع المدخلات: @f0/@f1 = ترتيب الملفات المرفوعة | @prev = مخرج السابق
  5343→      | @all_pdfs / @all_images / @all_cad
  5344→    النتيجة: ZIP يحتوي كل الملفات الناتجة + manifest.json
  5345→    """
  5346→    try:
  5347→        step_list = json.loads(steps)
  5348→    except Exception:
  5349→        raise HTTPException(400, "steps غير صالح")
  5350→
  5351→    with tempfile.TemporaryDirectory() as td:
  5352→        tdp = Path(td)
  5353→        in_dir, out_dir = tdp / "in", tdp / "out"
  5354→        in_dir.mkdir(); out_dir.mkdir()
  5355→
  5356→        uploaded: list[Path] = []
  5357→        for f in files:
  5358→            uploaded.append(_save_upload(f, in_dir))
  5359→
  5360→        def resolve(refs):
  5361→            paths = []
  5362→            for r in refs or []:
  5363→                if r == "@prev":
  5364→                    if not step_outputs:
  5365→                        raise HTTPException(400, "لا توجد خطوة سابقة")
  5366→                    paths.append(list(step_outputs.values())[-1])
  5367→                elif r.startswith("@f") and r[2:].isdigit():
  5368→                    idx = int(r[2:])
  5369→                    if idx >= len(uploaded):
  5370→                        raise HTTPException(400, f"مرجع ملف غير موجود: {r}")
  5371→                    paths.append(uploaded[idx])
  5372→                elif r == "@all_pdfs":
  5373→                    paths.extend(p for p in uploaded if p.suffix.lower() == ".pdf")
  5374→                elif r == "@all_images":
  5375→                    paths.extend(p for p in uploaded if p.suffix.lower() in (".png", ".jpg", ".jpeg", ".gif", ".webp", ".bmp"))
  5376→                elif r == "@all_cad":
  5377→                    paths.extend(p for p in uploaded if p.suffix.lower() in (".dxf", ".dwg", ".rvt"))
  5378→                elif r == "@all":
  5379→                    paths.extend(uploaded)
  5380→                else:
  5381→                    raise HTTPException(400, f"مرجع غير معروف: {r}")
  5382→            return paths
  5383→
  5384→        step_outputs: dict[str, Path] = {}
  5385→        results = []
  5386→        t0 = time.time()
  5387→
  5388→        for i, step in enumerate(step_list):
  5389→            op = step.get("op", "")
  5390→            params = step.get("params", {}) or {}
  5391→            paths = resolve(step.get("inputs") or ["@all"])
  5392→            if not paths:
  5393→                raise HTTPException(400, f"الخطوة {i+1}: لا توجد مدخلات")
  5394→            t1 = time.time()
  5395→            try:
  5396→                outs = run_op(op, paths, params, out_dir)
  5397→            except OpError as e:
  5398→                raise HTTPException(400, f"الخطوة {i+1} ({op}): {e}")
  5399→            except Exception as e:
  5400→                raise HTTPException(500, f"الخطوة {i+1} ({op}): {type(e).__name__}: {e}")
  5401→            step_outputs[f"step{i}"] = outs[-1]
  5402→            results.append({
  5403→                "step": i + 1, "op": op,
  5404→                "label": step.get("label", op),
  5405→                "duration": round(time.time() - t1, 2),
  5406→                "outputs": [{"name": o.name, "size": o.stat().st_size} for o in outs],
  5407→            })
  5408→
  5409→        # حزم كل المخرجات في ZIP
  5410→        buf = io.BytesIO()
  5411→        with zipfile.ZipFile(buf, "w", zipfile.ZIP_DEFLATED) as z:
  5412→            for o in sorted(out_dir.rglob("*")):
  5413→                if o.is_file():
  5414→                    z.write(o, o.relative_to(out_dir))
  5415→            z.writestr("manifest.json", json.dumps({
  5416→                "ok": True,
  5417→                "total_duration": round(time.time() - t0, 2),
  5418→                "results": results,
  5419→            }, ensure_ascii=False, indent=2))
  5420→        buf.seek(0)
  5421→        return Response(
  5422→            content=buf.read(),
  5423→            media_type="application/zip",
  5424→            headers={"Content-Disposition": "attachment; filename=pdf-station-results.zip"},
  5425→        )
