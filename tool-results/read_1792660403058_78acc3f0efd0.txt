    25→    25→/** نص عربي → PNG شفاف عبر Canvas (تشكيل حروف مثالي بواسطة المتصفح) */
    26→    26→async function arabicTextPng(
    27→    27→  text: string,
    28→    28→  opts: { size: number; color: string; bold?: boolean }
    29→    29→): Promise<{ bytes: Uint8Array; w: number; h: number }> {
    30→    30→  const font = `${opts.bold ? "700" : "500"} ${opts.size}px Tajawal, "Noto Sans Arabic", sans-serif`;
    31→    31→  try {
    32→    32→    await document.fonts.load(font, text);
    33→    33→  } catch {
    34→    34→    /* الخط غير متاح — نكمل بخط النظام */
    35→    35→  }
    36→    36→  const canvas = document.createElement("canvas");
    37→    37→  const ctx = canvas.getContext("2d")!;
    38→    38→  ctx.font = font;
    39→    39→  const metrics = ctx.measureText(text);
    40→    40→  const pad = Math.ceil(opts.size * 0.25);
    41→    41→  canvas.width = Math.ceil(metrics.width) + pad * 2;
    42→    42→  canvas.height = Math.ceil(opts.size * 1.45);
    43→    43→  const ctx2 = canvas.getContext("2d")!;
    44→    44→  ctx2.clearRect(0, 0, canvas.width, canvas.height);
    45→    45→  ctx2.font = font;
    46→    46→  ctx2.textAlign = "center";
    47→    47→  ctx2.textBaseline = "middle";
    48→    48→  ctx2.fillStyle = opts.color;
    49→    49→  ctx2.direction = "rtl";
    50→    50→  ctx2.fillText(text, canvas.width / 2, canvas.height / 2);
    51→    51→  const dataUrl = canvas.toDataURL("image/png");
    52→    52→  const res = await fetch(dataUrl);
    53→    53→  const buf = new Uint8Array(await res.arrayBuffer());
    54→    54→  return { bytes: buf, w: canvas.width, h: canvas.height };
    55→    55→}

    57→    57→function resultFile(name: string, bytes: Uint8Array | Blob, fromStep: string, kind: ResultFile["kind"]): ResultFile {
    58→    58→  const blob = bytes instanceof Blob ? bytes : new Blob([bytes.buffer as ArrayBuffer], { type: mimeOf(name) });
    59→    59→  return {
    60→    60→    id: `${Date.now()}-${Math.random().toString(36).slice(2, 7)}`,
    61→    61→    name,
    62→    62→    blob,
    63→    63→    url: URL.createObjectURL(blob),
    64→    64→    size: blob.size,
    65→    65→    fromStep,
    66→    66→    kind,
    67→    67→  };
    68→    68→}
    69→    69→
    70→    70→function mimeOf(name: string): string {
    71→    71→  if (name.endsWith(".pdf")) return "application/pdf";
    72→    72→  if (name.endsWith(".png")) return "image/png";
    73→    73→  if (name.endsWith(".jpg") || name.endsWith(".jpeg")) return "image/jpeg";
    74→    74→  if (name.endsWith(".txt")) return "text/plain;charset=utf-8";
    75→    75→  if (name.endsWith(".zip")) return "application/zip";
    76→    76→  return "application/octet-stream";
    77→    77→}

    97→    97→export async function runOp(
    98→    98→  op: string,
    99→    99→  inputs: EngineInput[],
   100→   100→  params: Record<string, string>,
   101→   101→  label: string,
   102→   102→  ctx: RunContext = {}
   103→   103→): Promise<ResultFile[]> {
   104→   104→  if (inputs.length === 0) {
   105→   105→    throw new Error("لا توجد ملفات مناسبة لهذه الخطوة — أضف ملفاتك أولًا");
   106→   106→  }
   107→   107→  switch (op) {
   108→   108→    case "merge":
   109→   109→      return opMerge(inputs, label, ctx);
   110→   110→    case "compress":
   111→   111→      return opCompress(inputs, params, label, ctx);
   112→   112→    case "extract":
   113→   113→      return opExtractOrDelete(inputs, params, "extract", label, ctx);
   114→   114→    case "delete":
   115→   115→      return opExtractOrDelete(inputs, params, "delete", label, ctx);
   116→   116→    case "rotate":
   117→   117→      return opRotate(inputs, params, label, ctx);
   118→   118→    case "reorder":
   119→   119→      return opReorder(inputs, params, label, ctx);
   120→   120→    case "split":
   121→   121→      return opSplit(inputs, label, ctx);
   122→   122→    case "watermark":
   123→   123→      return opWatermark(inputs, params, label, ctx);
   124→   124→    case "page_numbers":
   125→   125→      return opPageNumbers(inputs, params, label, ctx);
   126→   126→    case "images_to_pdf":
   127→   127→      return opImagesToPdf(inputs, params, label, ctx);
   128→   128→    case "pdf_to_images":
   129→   129→      return opPdfToImages(inputs, params, label, ctx);
   130→   130→    case "extract_text":
   131→   131→      return opExtractText(inputs, label, ctx);
   132→   132→    case "metadata":
   133→   133→      return opMetadata(inputs, params, label, ctx);
   134→   134→    case "nup":
   135→   135→      return opNup(inputs, params, label, ctx);
   136→   136→    default:
   137→   137→      throw new Error("عملية غير معروفة");
   138→   138→  }
   139→   139→}

   141→   141→/* دمج */
   142→   142→async function opMerge(inputs: EngineInput[], label: string, ctx: RunContext): Promise<ResultFile[]> {
   143→   143→  ctx.onProgress?.(`جارٍ دمج ${inputs.length} ملفات…`);
   144→   144→  const out = await PDFDocument.create();
   145→   145→  for (const input of inputs) {
   146→   146→    const src = await loadPdfLib(await bytesOf(input));
   147→   147→    const pages = await out.copyPages(src, src.getPageIndices());
   148→   148→    pages.forEach((p) => out.addPage(p));
   149→   149→  }
   150→   150→  const bytes = await out.save({ useObjectStreams: true });
   151→   151→  return [resultFile("الملف-المدموج.pdf", bytes, label, "pdf")];
   152→   152→}
   153→   153→
   154→   154→/* تصغير الحجم — ثلاثة مستويات بذكاء:
   155→   155→   ملف نصي → ضغط الهيكل فقط (الترقيم الصوري يكبّر النصي)
   156→   156→   ملف ممسوح → إعادة بناء الصفحات كصور JPEG (توفير حقيقي) */
   157→   157→async function hasTextLayer(bytes: Uint8Array): Promise<boolean> {
   158→   158→  try {
   159→   159→    const pdfjs = await getPdfjs();
   160→   160→    const doc = await pdfjs.getDocument({ data: bytes.slice() }).promise;
   161→   161→    let len = 0;
   162→   162→    for (let i = 1; i <= Math.min(2, doc.numPages); i++) {
   163→   163→      const page = await doc.getPage(i);
   164→   164→      const tc = await page.getTextContent();
   165→   165→      len += tc.items.reduce((acc, it) => acc + ("str" in it ? it.str.trim().length : 0), 0);
   166→   166→    }
   167→   167→    return len >= 20;
   168→   168→  } catch {
   169→   169→    return false;
   170→   170→  }
   171→   171→}

   173→   173→async function opCompress(
   174→   174→  inputs: EngineInput[],
   175→   175→  params: Record<string, string>,
   176→   176→  label: string,
   177→   177→  ctx: RunContext
   178→   178→): Promise<ResultFile[]> {
   179→   179→  const level = params.level ?? "متوسط";
   180→   180→  const results: ResultFile[] = [];
   181→   181→  for (const input of inputs) {
   182→   182→    ctx.onProgress?.(`جارٍ تصغير حجم «${input.name}»…`);
   183→   183→    const bytes = await bytesOf(input);
   184→   184→    const textual = await hasTextLayer(bytes);
   185→   185→    if (!textual) {
   186→   186→      // ملف ممسوح: إعادة بناء بجودة حسب المستوى
   187→   187→      const pdfjs = await getPdfjs();
   188→   188→      const src = await pdfjs.getDocument({ data: bytes.slice() }).promise;
   189→   189→      const out = await PDFDocument.create();
   190→   190→      const scale = level === "متوسط" ? 1.5 : 1.05;
   191→   191→      const quality = level === "متوسط" ? 0.8 : 0.55;
   192→   192→      for (let i = 1; i <= src.numPages; i++) {
   193→   193→        ctx.onProgress?.(`إعادة بناء الصفحة ${i} من ${src.numPages}…`);
   194→   194→        const page = await src.getPage(i);
   195→   195→        const viewport = page.getViewport({ scale });
   196→   196→        const canvas = document.createElement("canvas");
   197→   197→        canvas.width = Math.floor(viewport.width);
   198→   198→        canvas.height = Math.floor(viewport.height);
   199→   199→        const c2 = canvas.getContext("2d")!;
   200→   200→        c2.fillStyle = "#ffffff";
   201→   201→        c2.fillRect(0, 0, canvas.width, canvas.height);
   202→   202→        await page.render({ canvas, canvasContext: c2, viewport } as never).promise;
   203→   203→        const jpegUrl = canvas.toDataURL("image/jpeg", quality);
   204→   204→        const res = await fetch(jpegUrl);
   205→   205→        const jpegBytes = new Uint8Array(await res.arrayBuffer());
   206→   206→        const img = await out.embedJpg(jpegBytes);
   207→   207→        const p = out.addPage([
   208→   208→          page.getViewport({ scale: 1 }).width,
   209→   209→          page.getViewport({ scale: 1 }).height,
   210→   210→        ]);
   211→   211→        p.drawImage(img, { x: 0, y: 0, width: p.getWidth(), height: p.getHeight() });
   212→   212→      }
   213→   213→      const saved = await out.save({ useObjectStreams: true });
   214→   214→      results.push(resultFile("بعد-تصغير-الحجم.pdf", saved, label, "pdf"));
   215→   215→    } else {
   216→   216→      // ملف نصي: ضغط الهيكل + تنظيف البيانات في المتوسط والقوي
   217→   217→      const doc = await loadPdfLib(bytes);
   218→   218→      if (level !== "خفيف") {
   219→   219→        doc.setTitle("");
   220→   220→        doc.setAuthor("");
   221→   221→        doc.setSubject("");
   222→   222→        doc.setKeywords([]);
   223→   223→      }
   224→   224→      const saved = await doc.save({ useObjectStreams: true });
   225→   225→      results.push(resultFile("بعد-تصغير-الحجم.pdf", saved, label, "pdf"));
   226→   226→    }
   227→   227→  }
   228→   228→  return results;
   229→   229→}

   231→   231→/* استخراج / حذف صفحات */
   232→   232→async function opExtractOrDelete(
   233→   233→  inputs: EngineInput[],
   234→   234→  params: Record<string, string>,
   235→   235→  mode: "extract" | "delete",
   236→   236→  label: string,
   237→   237→  ctx: RunContext
   238→   238→): Promise<ResultFile[]> {
   239→   239→  const results: ResultFile[] = [];
   240→   240→  for (const input of inputs) {
   241→   241→    ctx.onProgress?.(`جارٍ ${mode === "extract" ? "استخراج" : "حذف"} الصفحات من «${input.name}»…`);
   242→   242→    const src = await loadPdfLib(await bytesOf(input));
   243→   243→    const total = src.getPageCount();
   244→   244→    const pages = parsePageRanges(params.pages ?? "", total);
   245→   245→    const keep =
   246→   246→      mode === "extract"
   247→   247→        ? pages.map((p) => p - 1)
   248→   248→        : Array.from({ length: total }, (_, i) => i + 1)
   249→   249→            .filter((p) => !pages.includes(p))
   250→   250→            .map((p) => p - 1);
   251→   251→    if (keep.length === 0) {
   252→   252→      throw new Error("لا توجد صفحات متبقية بعد هذه العملية — راجع الأرقام المختارة");
   253→   253→    }
   254→   254→    const out = await PDFDocument.create();
   255→   255→    const copied = await out.copyPages(src, keep);
   256→   256→    copied.forEach((p) => out.addPage(p));
   257→   257→    const saved = await out.save({ useObjectStreams: true });
   258→   258→    results.push(
   259→   259→      resultFile(mode === "extract" ? "الصفحات-المستخرجة.pdf" : "بعد-حذف-الصفحات.pdf", saved, label, "pdf")
   260→   260→    );
   261→   261→  }
   262→   262→  return results;
   263→   263→}

   266→   266→async function opRotate(
   267→   267→  inputs: EngineInput[],
   268→   268→  params: Record<string, string>,
   269→   269→  label: string,
   270→   270→  ctx: RunContext
   271→   271→): Promise<ResultFile[]> {
   272→   272→  const angle = Number(params.angle ?? "90");
   273→   273→  const results: ResultFile[] = [];
   274→   274→  for (const input of inputs) {
   275→   275→    ctx.onProgress?.(`جارٍ تدوير «${input.name}»…`);
   276→   276→    const doc = await loadPdfLib(await bytesOf(input));
   277→   277→    const total = doc.getPageCount();
   278→   278→    const pages = params.pages ? parsePageRanges(params.pages, total) : null;
   279→   279→    doc.getPages().forEach((page, idx) => {
   280→   280→      if (pages && !pages.includes(idx + 1)) return;
   281→   281→      const current = page.getRotation().angle;
   282→   282→      page.setRotation(degrees((current + angle) % 360));
   283→   283→    });
   284→   284→    const saved = await doc.save({ useObjectStreams: true });
   285→   285→    results.push(resultFile("بعد-التدوير.pdf", saved, label, "pdf"));
   286→   286→  }
   287→   287→  return results;
   288→   288→}

   312→   312→    results.push(resultFile("بعد-الترتيب.pdf", saved, label, "pdf"));

   318→   318→async function opSplit(inputs: EngineInput[], label: string, ctx: RunContext): Promise<ResultFile[]> {
   319→   319→  const results: ResultFile[] = [];
   320→   320→  const prefix = inputs.length > 1; // عدة ملفات معًا → نسبق الأسماء ب اسم الملف لتفادي التكرار
   321→   321→  for (const input of inputs) {
   322→   322→    ctx.onProgress?.(`جارٍ فصل صفحات «${input.name}»…`);
   323→   323→    const src = await loadPdfLib(await bytesOf(input));
   324→   324→    const total = src.getPageCount();
   325→   325→    const base = prefix ? input.name.replace(/\.pdf$/i, "") + "-" : "";
   326→   326→    for (let i = 0; i < total; i++) {
   327→   327→      ctx.onProgress?.(`فصل «${input.name}»: الصفحة ${toHindiDigits(i + 1)} من ${toHindiDigits(total)}`);
   328→   328→      const out = await PDFDocument.create();
   329→   329→      const [page] = await out.copyPages(src, [i]);
   330→   330→      out.addPage(page);
   331→   331→      const saved = await out.save({ useObjectStreams: true });
   332→   332→      results.push(resultFile(`${base}صفحة-${toHindiDigits(i + 1)}.pdf`, saved, label, "pdf"));
   333→   333→    }
   334→   334→  }
   335→   335→  return results;
   336→   336→}

   338→   338→/* ختم (علامة مائية) */
   339→   339→async function opWatermark(
   340→   340→  inputs: EngineInput[],
   341→   341→  params: Record<string, string>,
   342→   342→  label: string,
   343→   343→  ctx: RunContext
   344→   344→): Promise<ResultFile[]> {
   345→   345→  const text = params.text || "مسودة";
   346→   346→  const colorMap: Record<string, string> = { رمادي: "#6b7280", احمر: "#dc2626", ازرق: "#2563eb" };
   347→   347→  const color = colorMap[params.color ?? "رمادي"] ?? "#6b7280";
   348→   348→  const { bytes: png, w, h } = await arabicTextPng(text, { size: 88, color, bold: true });
   349→   349→  const results: ResultFile[] = [];
   350→   350→  for (const input of inputs) {
   351→   351→    ctx.onProgress?.(`جارٍ ختم «${input.name}»…`);
   352→   352→    const doc = await loadPdfLib(await bytesOf(input));
   353→   353→    const img = await doc.embedPng(png);
   354→   354→    for (const page of doc.getPages()) {
   355→   355→      const pw = page.getWidth();
   356→   356→      const ph = page.getHeight();
   357→   357→      const targetW = pw * 0.62;
   358→   358→      const targetH = (h / w) * targetW;
   359→   359→      const theta = (45 * Math.PI) / 180;
   360→   360→      const cx = pw / 2;
   361→   361→      const cy = ph / 2;
   362→   362→      const ox = (targetW / 2) * Math.cos(theta) - (targetH / 2) * Math.sin(theta);
   363→   363→      const oy = (targetW / 2) * Math.sin(theta) + (targetH / 2) * Math.cos(theta);
   364→   364→      page.drawImage(img, {
   365→   365→        x: cx - ox,
   366→   366→        y: cy - oy,
   367→   367→        width: targetW,
   368→   368→        height: targetH,
   369→   369→        rotate: degrees(45),
   370→   370→        opacity: 0.16,
   371→   371→      });
   372→   372→    }
   373→   373→    const saved = await doc.save({ useObjectStreams: true });
   374→   374→    results.push(resultFile("بعد-الختم.pdf", saved, label, "pdf"));
   375→   375→  }
   376→   376→  return results;
   377→   377→}

   379→   379→/* ترقيم الصفحات */
   380→   380→async function opPageNumbers(
   381→   381→  inputs: EngineInput[],
   382→   382→  params: Record<string, string>,
   383→   383→  label: string,
   384→   384→  ctx: RunContext
   385→   385→): Promise<ResultFile[]> {
   386→   386→  const results: ResultFile[] = [];
   387→   387→  for (const input of inputs) {
   388→   388→    ctx.onProgress?.(`جارٍ ترقيم «${input.name}»…`);
   389→   389→    const doc = await loadPdfLib(await bytesOf(input));
   390→   390→    const total = doc.getPageCount();
   391→   391→    const hindi = (params.digits ?? "هندي") === "هندي";
   392→   392→    const fmt = params.format ?? "رقم فقط";
   393→   393→    for (let i = 1; i <= total; i++) {
   394→   394→      const shown = hindi ? toHindiDigits(i) : String(i);
   395→   395→      const text = fmt === "صفحة من" ? `صفحة ${shown} من ${hindi ? toHindiDigits(total) : total}` : shown;
   396→   396→      const { bytes: png, w, h } = await arabicTextPng(text, { size: 60, color: "#1b2438" });
   397→   397→      const img = await doc.embedPng(png);
   398→   398→      const page = doc.getPage(i - 1);
   399→   399→      const targetH = 22;
   400→   400→      const targetW = (w / h) * targetH;
   401→   401→      page.drawImage(img, {
   402→   402→        x: (page.getWidth() - targetW) / 2,
   403→   403→        y: 16,
   404→   404→        width: targetW,
   405→   405→        height: targetH,
   406→   406→        opacity: 0.85,
   407→   407→      });
   408→   408→    }

   416→   416→async function opImagesToPdf(
   417→   417→  inputs: EngineInput[],
   418→   418→  params: Record<string, string>,
   419→   419→  label: string,
   420→   420→  ctx: RunContext
   421→   421→): Promise<ResultFile[]> {
   422→   422→  ctx.onProgress?.(`جارٍ تحويل ${inputs.length} صور إلى PDF…`);
   423→   423→  const out = await PDFDocument.create();

   425→   425→    const bytes = await bytesOf(input);

   431→   431→      const page = out.addPage([595.28, 841.89]);

   433→   433→      const maxW = page.getWidth() - margin * 2;
   434→   434→      const maxH = page.getHeight() - margin * 2;
   435→   435→      const scale = Math.min(maxW / img.width, maxH / img.height);
   436→   436→      page.drawImage(img, {
   437→   437→        x: (page.getWidth() - img.width * scale) / 2,
   438→   438→        y: (page.getHeight() - img.height * scale) / 2,
   439→   439→        width: img.width * scale,
   440→   440→        height: img.height * scale,
   441→   441→      });
   442→   442→    } else {
   443→   443→      const page = out.addPage([img.width, img.height]);
   444→   444→      page.drawImage(img, { x: 0, y: 0, width: img.width, height: img.height });
   445→   445→    }
   446→   446→  }
   447→   447→  const saved = await out.save({ useObjectStreams: true });
   448→   448→  return [resultFile("من-الصور.pdf", saved, label, "pdf")];

   452→   452→async function opPdfToImages(
   453→   453→  inputs: EngineInput[],
   454→   454→  params: Record<string, string>,
   455→   455→  label: string,
   456→   456→  ctx: RunContext
   457→   457→): Promise<ResultFile[]> {
   458→   458→  const pdfjs = await getPdfjs();
   459→   459→  const format = params.format ?? "png";
   460→   460→  const scaleMap: Record<string, number> = { مضغوطة: 1.2, متوازنة: 2.0, عالية: 3.0 };
   461→   461→  const scale = scaleMap[params.quality ?? "متوازنة"] ?? 2.0;
   462→   462→  const results: ResultFile[] = [];
   463→   463→  for (const input of inputs) {
   464→   464→    const bytes = await bytesOf(input);
   465→   465→    const src = await pdfjs.getDocument({ data: bytes.slice() }).promise;
   466→   466→    for (let i = 1; i <= src.numPages; i++) {
   467→   467→      ctx.onProgress?.(`تحويل الصفحة ${i} من ${src.numPages} إلى صورة…`);
   468→   468→      const page = await src.getPage(i);
   469→   469→      const viewport = page.getViewport({ scale });
   470→   470→      const canvas = document.createElement("canvas");
   471→   471→      canvas.width = Math.floor(viewport.width);
   472→   472→      canvas.height = Math.floor(viewport.height);
   473→   473→      const c2 = canvas.getContext("2d")!;
   474→   474→      c2.fillStyle = "#ffffff";
   475→   475→      c2.fillRect(0, 0, canvas.width, canvas.height);
   476→   476→      await page.render({ canvas, canvasContext: c2, viewport } as never).promise;
   477→   477→      const mime = format === "jpeg" ? "image/jpeg" : "image/png";
   478→   478→      const blob = await new Promise<Blob>((resolve, reject) =>
   479→   479→        canvas.toBlob((b) => (b ? resolve(b) : reject(new Error("تعذر إنشاء الصورة"))), mime, 0.9)
   480→   480→      );
   481→   481→      const ext = format === "jpeg" ? "jpg" : "png";
   482→   482→      results.push(resultFile(`${input.name.replace(/\.pdf$/i, "")}-صفحة-${i}.${ext}`, blob, label, "image"));
   483→   483→    }
   484→   484→  }
   485→   485→  return results;
   486→   486→}

   489→   489→async function opExtractText(inputs: EngineInput[], label: string, ctx: RunContext): Promise<ResultFile[]> {
   490→   490→  const pdfjs = await getPdfjs();
   491→   491→  const results: ResultFile[] = [];
   492→   492→  for (const input of inputs) {
   493→   493→    ctx.onProgress?.(`جارٍ استخراج نصوص «${input.name}»…`);
   494→   494→    const bytes = await bytesOf(input);
   495→   495→    const src = await pdfjs.getDocument({ data: bytes.slice() }).promise;
   496→   496→    const chunks: string[] = [];
   497→   497→    for (let i = 1; i <= src.numPages; i++) {
   498→   498→      const page = await src.getPage(i);
   499→   499→      const content = await page.getTextContent();
   500→   500→      const line = content.items
   501→   501→        .map((it) => ("str" in it ? it.str : ""))
   502→   502→        .join(" ")
   503→   503→        .replace(/\s+/g, " ")
   504→   504→        .trim();
   505→   505→      chunks.push(`── الصفحة ${toHindiDigits(i)} ──\n${line}`);
   506→   506→    }
   507→   507→    const text = chunks.join("\n\n");
   508→   508→    const blob = new Blob(["\ufeff" + text], { type: "text/plain;charset=utf-8" });
   509→   509→    results.push(resultFile("نص-الملف.txt", blob, label, "other"));
   510→   510→  }
   511→   511→  return results;
   512→   512→}

   576→   576→/** فحص PDF سريع: عدد الصفحات + مصغّرة + هل هو ممسوح ضوئيًا */
   577→   577→export async function inspectPdf(file: File): Promise<InspectInfo> {
   578→   578→  const pdfjs = await getPdfjs();
   579→   579→  const bytes = new Uint8Array(await file.arrayBuffer());
   580→   580→  const doc = await pdfjs.getDocument({ data: bytes.slice() }).promise;
   581→   581→  const pages = doc.numPages;
   582→   582→
   583→   583→  // مصغّرة الصفحة الأولى
   584→   584→  const page = await doc.getPage(1);
   585→   585→  const viewport = page.getViewport({ scale: 0.4 });
   586→   586→  const canvas = document.createElement("canvas");
   587→   587→  canvas.width = Math.floor(viewport.width);
   588→   588→  canvas.height = Math.floor(viewport.height);
   589→   589→  const c2 = canvas.getContext("2d")!;
   590→   590→  c2.fillStyle = "#ffffff";
   591→   591→  c2.fillRect(0, 0, canvas.width, canvas.height);
   592→   592→  await page.render({ canvas, canvasContext: c2, viewport } as never).promise;
   593→   593→  const thumb = canvas.toDataURL("image/png");
   594→   594→
   595→   595→  // كشف الممسوح: نص الصفحتين الأوليين
   596→   596→  let textLen = 0;
   597→   597→  for (let i = 1; i <= Math.min(2, pages); i++) {
   598→   598→    const p = await doc.getPage(i);
   599→   599→    const tc = await p.getTextContent();
   600→   600→    textLen += tc.items.reduce((acc, it) => acc + ("str" in it ? it.str.trim().length : 0), 0);
   601→   601→  }
   602→   602→  return { pages, thumb, scanned: textLen < 20 };
   603→   603→}

   606→   606→export async function inspectImage(file: File): Promise<{ thumb: string }> {
   607→   607→  return new Promise((resolve) => {
   608→   608→    const url = URL.createObjectURL(file);
   609→   609→    const img = new Image();
   610→   610→    img.onload = () => {
   611→   611→      const canvas = document.createElement("canvas");
   612→   612→      const scale = Math.min(1, 220 / img.width);
   613→   613→      canvas.width = Math.floor(img.width * scale);
   614→   614→      canvas.height = Math.floor(img.height * scale);
   615→   615→      canvas.getContext("2d")!.drawImage(img, 0, 0, canvas.width, canvas.height);
   616→   616→      URL.revokeObjectURL(url);
   617→   617→      resolve({ thumb: canvas.toDataURL("image/png") });
   618→   618→    };
   619→   619→    img.onerror = () => {
   620→   620→      URL.revokeObjectURL(url);
   621→   621→      resolve({ thumb: "" });
   622→   622→    };
   623→   623→    img.src = url;
   624→   624→  });
   625→   625→}

   650→   650→export async function zipResults(files: ResultFile[]): Promise<Blob> {
   651→   651→  const JSZip = (await import("jszip")).default;
   652→   652→  const zip = new JSZip();
   653→   653→  const used = new Set<string>();
   654→   654→  for (const f of files) {
   655→   655→    let name = f.name;
   656→   656→    let k = 2;
   657→   657→    while (used.has(name)) {
   658→   658→      name = f.name.replace(/(\.[^.]+)$/, `-${k}$1`);
   659→   659→      k++;
   660→   660→    }
   661→   661→    used.add(name);
   662→   662→    zip.file(name, f.blob);
   663→   663→  }
   664→   664→  return zip.generateAsync({ type: "blob", compression: "DEFLATE" });
   665→   665→}

   668→   668→export function downloadBlob(blob: Blob, name: string): boolean {
   669→   669→  const url = URL.createObjectURL(blob);
   670→   670→  const a = document.createElement("a");
   671→   671→  a.href = url;
   672→   672→  a.download = name;
   673→   673→  a.rel = "noopener";
   674→   674→  document.body.appendChild(a);
   675→   675→  a.click();
   676→   676→  a.remove();
   677→   677→  // مهلة أطول حتى يبقى الرابط صالحًا إن احتاج المستخدم فتحه من تبويب مستقل
   678→   678→  setTimeout(() => URL.revokeObjectURL(url), 60_000);
   679→   679→  return isEmbedded();
   680→   680→}
