   114→async function renderPageDataUrl(file: File, pageNo: number, maxW = 820): Promise<string> {
   115→  const pdfjs = await import("pdfjs-dist");
   116→  pdfjs.GlobalWorkerOptions.workerSrc = "/pdf.worker.min.mjs";
   117→  const bytes = new Uint8Array(await file.arrayBuffer());
   118→  const doc = await pdfjs.getDocument({ data: bytes }).promise;
   119→  const page = await doc.getPage(Math.max(1, pageNo));
   120→  const v1 = page.getViewport({ scale: 1 });
   121→  const viewport = page.getViewport({ scale: Math.min(1.8, maxW / v1.width) });
   122→  const canvas = document.createElement("canvas");
   123→  canvas.width = Math.floor(viewport.width);
   124→  canvas.height = Math.floor(viewport.height);
   125→  const c2 = canvas.getContext("2d")!;
   126→  c2.fillStyle = "#ffffff";
   127→  c2.fillRect(0, 0, canvas.width, canvas.height);
   128→  await page.render({ canvas, canvasContext: c2, viewport } as never).promise;
   129→  return canvas.toDataURL("image/jpeg", 0.85);
   130→}

   132→function fmtSize(b: number): string {
   133→  if (b >= 1048576) return `${toHindiDigits((b / 1048576).toFixed(1).replace(".", "٫"))} م.ب`;
   134→  return `${toHindiDigits(Math.max(1, Math.round(b / 1024)))} ك.ب`;
   135→}

   137→function extLabel(name: string): string {
   138→  const e = name.match(/\.([^.]+)$/)?.[1]?.toUpperCase() ?? "ملف";
   139→  return e === "JPG" ? "JPG" : e;
   140→}

   144→export function Workbench() {
   145→  const files = useWorkspace((s) => s.files);
   146→  const results = useWorkspace((s) => s.results);
   147→  const runState = useWorkspace((s) => s.runState);
   148→  const runMessage = useWorkspace((s) => s.runMessage);
   149→  const runProgress = useWorkspace((s) => s.runProgress);
   150→  const addFiles = useWorkspace((s) => s.addFiles);
   151→  const removeFile = useWorkspace((s) => s.removeFile);
   152→  const runOnFiles = useWorkspace((s) => s.runOnFiles);
   153→  const renameResult = useWorkspace((s) => s.renameResult);
   154→  const clearResults = useWorkspace((s) => s.clearResults);
   155→  const downloadAll = useWorkspace((s) => s.downloadAll);
   156→  const addEditorResult = useWorkspace((s) => s.addEditorResult);
   157→
   158→  const [activeId, setActiveId] = useState<string | null>(null);
   159→  const [selFiles, setSelFiles] = useState<Set<string> | null>(null);
   160→  const [tool, setTool] = useState<OpId | null>(null);
   161→  const [params, setParams] = useState<Record<string, string>>({});
   162→  const [selPages, setSelPages] = useState<number[]>([]);
   163→  const [thumbs, setThumbs] = useState<Record<string, string[]>>({});
   164→  const [page, setPage] = useState(1);
   165→
   166→  const [renaming, setRenaming] = useState<{ id: string; value: string } | null>(null);
   167→  const [dragOver, setDragOver] = useState(false);
   168→  const [editing, setEditing] = useState(false);
   169→  const inputRef = useRef<HTMLInputElement>(null);
   170→  const thumbsReq = useRef<Set<string>>(new Set());
   171→
   172→  /* الملف النشط: المعيّن بالنقر أو آخر ملف أُضيف — اشتقاق بلا effects */
   173→  const activeFile = files.find((f) => f.id === activeId) ?? files[files.length - 1] ?? null;
   174→  const effectiveSel = selFiles ?? (activeFile ? new Set([activeFile.id]) : new Set<string>());
   175→  const running = runState === "running";
   176→
   177→  /* مصغرات صفحات الـ PDF النشط */
   178→  useEffect(() => {
   179→    if (!activeFile || activeFile.kind !== "pdf") return;
   180→    if (thumbs[activeFile.id] || thumbsReq.current.has(activeFile.id)) return;
   181→    thumbsReq.current.add(activeFile.id);
   182→    pdfPageThumbs(activeFile.file, 60)
   183→      .then((arr) => setThumbs((t) => ({ ...t, [activeFile.id]: arr })))
   184→      .catch(() => thumbsReq.current.delete(activeFile.id));
   185→  }, [activeFile, thumbs]);
   186→
   187→  /* المعاينة الكبيرة للصفحة النشطة — مفتوحة بالمفتاح لتفادي setState المتزامن */
   188→  const previewKey = activeFile ? `${activeFile.id}:${activeFile.kind === "image" ? "img" : page}` : "";
   189→  const [previewState, setPreviewState] = useState<{ key: string; url: string } | null>(null);
   190→  useEffect(() => {
   191→    if (!activeFile || !previewKey) return;
   192→    let alive = true;
   193→    let objUrl: string | null = null;
   194→    const done = (url: string) => {
   195→      if (alive) setPreviewState({ key: previewKey, url });
   196→    };
   197→    if (activeFile.kind === "image") {
   198→      Promise.resolve().then(() => {
   199→        objUrl = URL.createObjectURL(activeFile.file);
   200→        done(objUrl);
   201→      });
   202→    } else {
   203→      renderPageDataUrl(activeFile.file, page)
   204→        .then(done)
   205→        .catch(() => undefined);
   206→    }
   207→    return () => {
   208→      alive = false;
   209→      if (objUrl) URL.revokeObjectURL(objUrl);
   210→    };
   211→  }, [activeFile, page, previewKey]);
   212→  const preview = previewState && previewState.key === previewKey ? previewState.url : "";
   213→
   214→  const editorEnabled =
   215→    activeFile?.kind === "pdf" || activeFile?.kind === "image";
   216→
   217→  const pickTool = (id: OpId) => {
   218→    setTool(id);
   219→    const def: Record<string, string> = {};
   220→    for (const f of TOOLS_BY_ID[id].fields) if (f.default !== undefined) def[f.key] = f.default;
   221→    const pf = TOOLS_BY_ID[id].fields.find((f) => f.pagesPicker);
   222→    if (pf) def[pf.key] = selPages.join(",");
   223→    setParams(def);
   224→  };
   225→
   226→  const togglePage = (n: number) => {
   227→    const next = selPages.includes(n)
   228→      ? selPages.filter((x) => x !== n)
   229→      : [...selPages, n].sort((a, b) => a - b);
   230→    setSelPages(next);
   231→    if (tool) {
   232→      const pf = TOOLS_BY_ID[tool].fields.find((f) => f.pagesPicker);
   233→      if (pf) setParams((p) => ({ ...p, [pf.key]: next.join(",") }));
   234→    }
   235→  };
   236→
   237→  const toggleSel = (id: string) =>
   238→    setSelFiles((s) => {
   239→      const n = new Set(s ?? (activeFile ? [activeFile.id] : []));
   240→      if (n.has(id)) {
   241→        if (n.size > 1) n.delete(id);
   242→      } else {
   243→        n.add(id);
   244→      }
   245→      return n;
   246→    });
   247→
   248→  const expectedName = useMemo(() => {
   249→    if (!tool || !activeFile) return null;
   250→    const t = TOOLS_BY_ID[tool];
   251→    const target = t.accepts.includes(activeFile.kind)
   252→      ? activeFile
   253→      : files.find((f) => effectiveSel.has(f.id) && t.accepts.includes(f.kind));
   254→    return target ? resultName(tool, target.name) : null;
   255→  }, [tool, activeFile, files, effectiveSel]);
   256→
   257→  /* ذكاء الأدوات: كل أداة تعمل فقط لو المحدد فيهم نوع تقبله */
   258→  const selKinds = useMemo(() => {
   259→    const ks = new Set<FileKind>();
   260→    for (const f of files) if (effectiveSel.has(f.id)) ks.add(f.kind);
   261→    return ks;
   262→  }, [files, effectiveSel]);
   263→
   264→  const toolEnabled = (id: OpId): boolean => {
   265→    const t = TOOLS_BY_ID[id];
   266→    if (!t) return false;
   267→    for (const k of selKinds) if (t.accepts.includes(k)) return true;
   268→    return false;
   269→  };
   270→
   271→  const hintDisabled = (id: OpId) => {
   272→    const t = TOOLS_BY_ID[id];
   273→    if (!t) return;
   274→    if (t.accepts.includes("image") && !t.accepts.includes("pdf"))
   275→      toast.info("الأداة دي للصور — حدّد صورة من قائمة الملفات");
   276→    else if (activeFile?.kind === "image")
   277→      toast.info("الأداة دي لملفات PDF — للصور استخدم «من صور»");
   278→    else if (activeFile && activeFile.kind !== "pdf")
   279→      toast.info("ندعم ملفات PDF والصور فقط حاليًا — أنواع أخرى غير مدعومة بعد");
   280→    else toast.info("حدّد ملف PDF من قائمة الملفات أولًا");
   281→  };
   282→
   283→  const editorSave = (name: string, blob: Blob) => {
   284→    addEditorResult(name, blob);
   285→    setEditing(false);
   286→    toast.success("حُفظت التعديلات — الملف جاهز للتنزيل");
   287→  };
   288→
   289→  const execute = async () => {
   290→    if (!tool || running) return;
   291→    const t = TOOLS_BY_ID[tool];
   292→    const chosen = files.filter((f) => effectiveSel.has(f.id) && t.accepts.includes(f.kind));
   293→    if (chosen.length === 0) {
   294→      toast.error(t.accepts.includes("image") ? "حدد صورًا من قائمة الملفات أولًا" : "حدد ملف PDF من قائمة الملفات أولًا");
   295→      return;
   296→    }
   297→    await runOnFiles(tool, params, chosen.map((f) => f.id));
   298→    const st = useWorkspace.getState();
   299→    if (st.runState === "error") toast.error(st.runMessage);
   300→  };
   301→
   302→  const download = (r: ResultFile) => {

   304→    if (embedded) {
   305→      toast("بدأ التنزيل", {
   306→        description: "إن لم يظهر الملف فافتح النتيجة في تبويب مستقل",
   307→        action: { label: "فتح في تبويب", onClick: () => openBlobInNewTab(r.blob) },
   308→        duration: 10_000,
   309→      });
   310→    } else {
   311→      toast.success("بدأ التنزيل");
   312→    }
   313→  };

   322→  return (

   325→      onDragOver={(e) => {
   326→        e.preventDefault();
   327→        setDragOver(true);
   328→      }}
   329→      onDragLeave={() => setDragOver(false)}

   331→        e.preventDefault();
   332→        setDragOver(false);
   333→        const fs = Array.from(e.dataTransfer.files);
   334→        if (fs.length) addFiles(fs);
   335→      }}
   336→    >
   337→      <input
   338→        ref={inputRef}
   339→        type="file"
   340→        multiple
   341→        accept=".pdf,image/*"
   342→        className="hidden"
   343→        onChange={(e) => {
   344→          const fs = Array.from(e.target.files ?? []);
   345→          if (fs.length) addFiles(fs);
   346→          e.target.value = "";
   347→        }}
   348→      />
   349→
   350→      {files.length === 0 ? (
   351→        /* ─── لا ملفات: منطقة إفلات واحدة ─── */
   352→        <div
   353→          className={cn(
   354→            "mt-8 rounded-3xl border-2 border-dashed bg-white/70 py-20 text-center transition",
   355→            dragOver ? "border-primary bg-primary/5" : "border-primary/30"

   357→          onClick={() => inputRef.current?.click()}
   358→          role="button"
   359→          tabIndex={0}
   360→          onKeyDown={(e) => e.key === "Enter" && inputRef.current?.click()}
   361→        >
   362→          <div className="mx-auto grid size-16 place-items-center rounded-2xl bg-primary text-primary-foreground shadow-lg shadow-primary/25">
   363→            <Plus className="size-7" aria-hidden />
   364→          </div>
   365→          <h2 className="mt-4 text-xl font-extrabold">اسحب ملفاتك هنا</h2>
   366→          <p className="mt-1 text-sm text-muted-foreground">PDF أو صور</p>

   368→            className="mt-5 px-8 font-extrabold"
   369→            onClick={(e) => {
   370→              e.stopPropagation();

   374→            اختيار ملفات
   375→          </Button>
   376→        </div>
   377→      ) : (
   378→        <>
   379→          {/* ─── شريط الأدوات ─── */}
   380→          <div className="flex flex-wrap gap-0.5 rounded-2xl border border-border bg-white p-1.5 shadow-sm">
   381→            {TOOLBAR.map(({ id, label }) => {
   382→              const isEditor = id === "editor";
   383→              const ok = isEditor ? editorEnabled : toolEnabled(id);

   415→          </div>
   416→
   417→          {/* ─── الجسم: ملفات + صفحات + معاينة ─── */}
   418→          <div className="mt-4 grid gap-4 lg:grid-cols-[200px_130px_1fr]">

   420→              <p className="px-1 text-[11px] font-bold text-muted-foreground">الملفات</p>
   421→              <div className="flex max-h-[460px] flex-col gap-2 overflow-y-auto">
