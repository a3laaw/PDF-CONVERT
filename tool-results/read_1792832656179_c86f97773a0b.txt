   115→   115→async function renderPageDataUrl(file: File, pageNo: number, maxW = 820): Promise<string> {
   116→   116→  const pdfjs = await import("pdfjs-dist");
   117→   117→  pdfjs.GlobalWorkerOptions.workerSrc = "/pdf.worker.min.mjs";
   118→   118→  const bytes = new Uint8Array(await file.arrayBuffer());
   119→   119→  const doc = await pdfjs.getDocument({ data: bytes }).promise;
   120→   120→  const page = await doc.getPage(Math.max(1, pageNo));
   121→   121→  const v1 = page.getViewport({ scale: 1 });
   122→   122→  const viewport = page.getViewport({ scale: Math.min(1.8, maxW / v1.width) });
   123→   123→  const canvas = document.createElement("canvas");
   124→   124→  canvas.width = Math.floor(viewport.width);
   125→   125→  canvas.height = Math.floor(viewport.height);
   126→   126→  const c2 = canvas.getContext("2d")!;
   127→   127→  c2.fillStyle = "#ffffff";
   128→   128→  c2.fillRect(0, 0, canvas.width, canvas.height);
   129→   129→  await page.render({ canvas, canvasContext: c2, viewport } as never).promise;
   130→   130→  return canvas.toDataURL("image/jpeg", 0.85);
   131→   131→}

   133→   133→function fmtSize(b: number): string {
   134→   134→  if (b >= 1048576) return `${toHindiDigits((b / 1048576).toFixed(1))} م.ب`;
   135→   135→  return `${toHindiDigits(Math.max(1, Math.round(b / 1024)))} ك.ب`;
   136→   136→}
